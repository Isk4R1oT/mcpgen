    ```typescript
    // apps/api/tests/e2e/_helpers/stripe-cli.ts
    //
    // E2E helper: shell out to `stripe trigger ...` via Bun.spawn when stripe-cli
    // is available locally. Used to drive real webhook events through the
    // `stripe listen --forward-to ...` forwarder running in Terminal 3.
    //
    // Falls back to direct webhook POST with synthesized event JSON + valid signature
    // computed via the Stripe SDK if STRIPE_CLI_AVAILABLE=0 (CI-like env without
    // stripe-cli installed).

    import Stripe from 'stripe';

    const HAS_CLI = process.env.STRIPE_CLI_AVAILABLE !== '0';

    export async function triggerStripeEvent(
      eventType: string,
      overrides: Record<string, unknown>,
    ): Promise<void> {
      if (HAS_CLI) {
        const proc = Bun.spawn(['stripe', 'trigger', eventType, '--add', `metadata.e2e=true`], {
          stdout: 'pipe', stderr: 'pipe',
        });
        const exit = await proc.exited;
        if (exit !== 0) throw new Error(`stripe trigger ${eventType} exit ${String(exit)}`);
        return;
      }
      // Fallback: synthesize event + POST to /api/v1/stripe/webhook with valid signature
      const stripe = new Stripe(process.env.STRIPE_SECRET_KEY!);
      const event: Stripe.Event = {
        id: `evt_e2e_${Date.now().toString(36)}`,
        type: eventType,
        data: { object: overrides },
        created: Math.floor(Date.now() / 1000),
        api_version: '2025-09-30.clover',
        livemode: false,
        object: 'event',
        pending_webhooks: 0,
        request: { id: null, idempotency_key: null },
      } as Stripe.Event;
      const payload = JSON.stringify(event);
      const sig = stripe.webhooks.generateTestHeaderString({
        payload, secret: process.env.STRIPE_WEBHOOK_SECRET!,
      });
      const res = await fetch('http://localhost:8787/api/v1/stripe/webhook', {
        method: 'POST',
        headers: { 'stripe-signature': sig, 'Content-Type': 'application/json' },
        body: payload,
      });
      if (!res.ok) throw new Error(`webhook POST failed: ${String(res.status)}`);
    }
    ```
