```sql
SELECT COALESCE(SUM(uh.call_count), 0)::int AS used
FROM usage_hourly uh
JOIN deployments d ON d.id = uh.deployment_id
JOIN generations g ON g.id = d.generation_id
JOIN projects p ON p.id = g.project_id
WHERE p.org_id = $1
  AND uh.event_type = $2
  AND uh.bucket >= $3
  AND uh.bucket < $4;
```

    ```typescript
    // apps/api/src/lib/quota.ts
    //
    // CTRL-07 / D-12: Pre-enqueue quota check.
    // Reads TimescaleDB usage_hourly aggregate (real-time quota truth per Pitfall #16);
    // returns 429-shaped result when used >= limit; PAYG plan never blocks.

    import { eq } from 'drizzle-orm';
    import { db } from '../db.js';
    import { organizations } from '@mcpgen/contracts/db-schema';
    import { QUOTA_LIMITS, type PlanTier } from '@mcpgen/contracts';
    import { getQuotaUsage } from './quota-queries.js';

    export type QuotaResult =
      | { ok: true; used: number; limit: number; reset_at: Date }
      | { ok: false; reason: 'quota_exceeded'; used: number; limit: number; reset_at: Date };

    export async function checkQuota(
      orgId: string,
      eventType: 'f3_eval' | 'generation',
    ): Promise<QuotaResult> {
      const org = await db.query.organizations.findFirst({ where: eq(organizations.id, orgId) });
      if (!org) throw new Error(`org_not_found: ${orgId}`);

      const tier = org.plan_tier as PlanTier;

      // PAYG plan never blocks.
      if (tier === 'payg') {
        return { ok: true, used: 0, limit: Infinity, reset_at: new Date() };
      }

      const limits = QUOTA_LIMITS[tier];
      if (!limits) throw new Error(`unknown_plan_tier: ${String(tier)}`);
      const limit = limits[eventType];

      const periodStart = new Date(org.quota_period_start);
      const periodEnd = new Date(periodStart);
      periodEnd.setMonth(periodEnd.getMonth() + 1);

      const used = await getQuotaUsage(orgId, eventType, periodStart, periodEnd);

      if (used >= limit) {
        return { ok: false, reason: 'quota_exceeded', used, limit, reset_at: periodEnd };
      }
      return { ok: true, used, limit, reset_at: periodEnd };
    }
    ```

    ```typescript
    export function quotaGate(eventType: 'f3_eval' | 'generation') {
    ```

    ```typescript
    // apps/api/src/inngest/functions/cost-cap-enforcer.ts
    //
    // CTRL-06 / D-13: Cost-cap enforcement (event-triggered, NOT cron).
    // Stable function ID: INNGEST_FUNCTION_IDS.COST_CAP_ENFORCER ('cost-cap-enforcer-v1').
    // Trigger: 'generation/cost.updated' Inngest event from sse-callback.ts.
    //
    // CRITICAL: concurrency: { limit: 1, key: 'event.data.jobId' } serializes processing
    // per-generation so two simultaneous cost events for the same job cannot both pass
    // the threshold check before either updates generations.status='cost_capped'.
    //
    // Algorithm:
    //   1. persist-cost: UPDATE generations.cumulative_cost_usd
    //   2. lookup-org: read plan_tier
    //   3. threshold-check: cumulativeCostUsd > cap (free 0.50 / pro 2.00)
    //   4. cancel-engine: POST {ENGINE_ENDPOINT}/internal/v1/cancel-generation with M2M Bearer
    //   5. mark-cost-capped: UPDATE generations.status='cost_capped'
    //   6. emit-billable-event: INSERT usage_events_outbox row for accumulated cost

    import { eq } from 'drizzle-orm';
    import { ulid } from 'ulid';
    import { INNGEST_FUNCTION_IDS, LAUNCH_CRITERIA, generations, organizations, usage_events_outbox } from '@mcpgen/contracts';
    import { SYNTHETIC_DEPLOYMENT_ID } from '@mcpgen/contracts/billing-types';  // B2 — single source of truth (Plan 01 Wave-1 seed)
    import { inngest } from '../client.js';
    import { db } from '../../db.js';
    import { getM2mTokenForEngine } from '../../lib/m2m-token.js';

    interface CostUpdateEvent {
      data: { jobId: string; passName: string; cumulativeCostUsd: number };
    }

    interface CostCapEnv {
      ENGINE_ENDPOINT: string;
      LOGTO_ENDPOINT: string;
      LOGTO_M2M_APP_ID: string;
      LOGTO_M2M_APP_SECRET: string;
      LOGTO_M2M_RESOURCE_INDICATOR: string;
    }

    export const costCapEnforcer = inngest.createFunction(
      {
        id: INNGEST_FUNCTION_IDS.COST_CAP_ENFORCER,
        concurrency: { limit: 1, key: 'event.data.jobId' },  // T-8-19 mitigation
      },
      { event: 'generation/cost.updated' },
      async ({ event, step, env }) => {
        const { jobId, cumulativeCostUsd } = (event as CostUpdateEvent).data;
        const e = env as CostCapEnv;

        await step.run('persist-cost', async () => {
          await db.update(generations)
            .set({ cumulative_cost_usd: String(cumulativeCostUsd) })
            .where(eq(generations.id, jobId));
        });

        const gen = await step.run('lookup-org', async () => {
          const row = await db.query.generations.findFirst({
            where: eq(generations.id, jobId),
            with: { project: { with: { organization: true } } },
          });
          return row;
        });

        const planTier = (gen as unknown as { project: { organization: { plan_tier: string } } } | undefined)?.project?.organization?.plan_tier ?? 'free';
        const cap = planTier === 'pro' ? LAUNCH_CRITERIA.COST_CAP_PRO_USD : LAUNCH_CRITERIA.COST_CAP_FREE_USD;
        // PAYG never caps — every overage bills.
        if (planTier === 'payg') return { capped: false, reason: 'payg_no_cap' };

        if (cumulativeCostUsd > cap) {
          await step.run('cancel-engine', async () => {
            const m2mToken = await getM2mTokenForEngine(e);
            const resp = await fetch(`${e.ENGINE_ENDPOINT}/internal/v1/cancel-generation`, {
              method: 'POST',
              headers: { 'Authorization': `Bearer ${m2mToken}`, 'Content-Type': 'application/json' },
              body: JSON.stringify({ job_id: jobId, reason: 'cost_cap_exceeded', cap_usd: cap }),
            });
            if (!resp.ok) throw new Error(`engine cancel failed: ${String(resp.status)}`);
          });

          await step.run('mark-cost-capped', async () => {
            await db.update(generations)
              .set({ status: 'cost_capped' })
              .where(eq(generations.id, jobId));
          });

          await step.run('emit-billable-event', async () => {
            await db.insert(usage_events_outbox).values({
              id: ulid(),
              deployment_id: SYNTHETIC_DEPLOYMENT_ID,  // B2 — same constant Plan 02 seeder uses; FK chain resolves to the synthetic deployment row
              event_type: 'generation_cost_capped',
              event_payload: { job_id: jobId, billed_cost_usd: cumulativeCostUsd, cap_usd: cap, plan_tier: planTier },
              idempotency_key: `${jobId}_cost_capped`,
            }).onConflictDoNothing();
          });

          return { capped: true, billed_cost_usd: cumulativeCostUsd, cap_usd: cap };
        }

        return { capped: false, cumulative_cost_usd: cumulativeCostUsd, cap_usd: cap };
      },
    );
    ```

    NOTE: The `usage_events_outbox.deployment_id` FK NOT NULL constraint requires a real `deployments` row for cost-capped events (no real deployment exists yet at cap-fire time). Use the SYNTHETIC_DEPLOYMENT_ID sentinel deployment row (created by `seed-synthetic-usage.ts` Plan 02 Task 3 + imported from `@mcpgen/contracts/billing-types` per B2). Both Plan 02 seeder + this enforcer reference the SAME constant — single source of truth eliminates the UUID drift that would otherwise break the FK at runtime.
