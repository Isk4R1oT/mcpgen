export const driftWatcherCheck = inngest.createFunction(
  {
    id: INNGEST_FUNCTION_IDS.DRIFT_WATCHER_CHECK,
    triggers: [{ event: 'drift/check.requested' }],
    retries: 3,
  },
  async ({ event, step }) => {
    const { deploymentId, specUrl } = event.data;
    if (!specUrl) return { skipped: 'no_source_url' };
    // Q1 graceful fallback: if engine returns 502/503/timeout → return {skipped: 'engine_unavailable'}
    const newIr = await step.run('parse-stage-a', async () => {
      const m2m = await getM2mTokenForEngine(env);
      const resp = await fetch(`${env.ENGINE_ENDPOINT}/internal/v1/parse`, {
        method: 'POST',
        headers: { 'Authorization': `Bearer ${m2m}`, 'Content-Type': 'application/json' },
        body: JSON.stringify({ spec_url: specUrl }),
        signal: AbortSignal.timeout(10_000),
      });
      if (resp.status === 502 || resp.status === 503) return null;  // engine not ready (Q1 fallback)
      if (!resp.ok) throw new Error(`engine parse failed: ${resp.status}`);
      const json = ParseResponse.parse(await resp.json());
      return json.raw_ir;
    });
    if (newIr === null) return { skipped: 'engine_unavailable' };
    // ... baseline diff + persist + email per RESEARCH §13
  },
);
```

const IGNORED_FIELDS = ['summary', 'description', 'tags', 'externalDocs'];
const X_PREFIX = 'x-';

function stripCosmetic(node: unknown): unknown {
  if (Array.isArray(node)) return node.map(stripCosmetic);
  if (node && typeof node === 'object') {
    const out: Record<string, unknown> = {};
    for (const [k, v] of Object.entries(node)) {
      if (IGNORED_FIELDS.includes(k) || k.startsWith(X_PREFIX)) continue;
      out[k] = stripCosmetic(v);
    }
    const sorted: Record<string, unknown> = {};
    for (const k of Object.keys(out).sort()) sorted[k] = out[k];
    return sorted;
  }
  return node;
}
```

    //
    // CTRL-07 / D-15 / Pitfall #16: Daily TimescaleDB ↔ Stripe Meters reconciliation.
    // Stable function ID: INNGEST_FUNCTION_IDS.USAGE_RECONCILER ('usage-reconciler-v1').
    // Schedule: 02:00 UTC daily.
    //
    // Quota truth = TimescaleDB usage_hourly (real-time).
    // Stripe Meters = billing eventual; this cron alerts when drift > 2.0%.
    //
    // References:
    //   - .planning/phases/08-auth-billing/08-RESEARCH.md §6 D-15, §11
    //   - .planning/phases/08-auth-billing/08-CONTEXT.md D-15

    import { ulid } from 'ulid';
    import { sql } from 'drizzle-orm';
    import {
      INNGEST_FUNCTION_IDS,
      reconciliation_log,
      organizations,
    } from '@mcpgen/contracts';
    import { cron } from 'inngest';
    import { inngest } from '../client.js';
    import { db } from '../../db.js';
    import { getStripe } from '../../lib/stripe-client.js';
    import { sendReconciliationAlert } from '../../lib/email/resend-client.js';

    const RECONCILIATION_DRIFT_THRESHOLD_PCT = 2.0;

    interface ReconcilerEnv {
      STRIPE_SECRET_KEY: string;
      STRIPE_METER_EVALS_ID: string;
      STRIPE_METER_TOOL_CALLS_ID: string;
      STRIPE_METER_GENERATIONS_ID: string;
      RESEND_API_KEY: string;
      OPS_EMAIL: string;
      LOGTO_BASE_URL: string;
      OPS_FROM_EMAIL?: string;
    }

    export const usageReconciler = inngest.createFunction(
      { id: INNGEST_FUNCTION_IDS.USAGE_RECONCILER, triggers: [cron('0 2 * * *')] },
      async ({ step, env }) => {
        // W3: typed `env` parameter from Inngest handler signature (PATTERNS.md §C);
        // matches Plan 03 cost-cap-enforcer style. NO `process.env` access in handler body
        // OR module top-level (METER_ID_BY_EVENT moved inside handler so STRIPE_METER_*_ID
        // reads from typed `env` parameter, not boot-time process.env capture).
        const typedEnv = env as ReconcilerEnv;
        const stripe = getStripe(typedEnv);

        const METER_ID_BY_EVENT: Record<string, string | undefined> = {
          f3_eval: typedEnv.STRIPE_METER_EVALS_ID,
          tool_call: typedEnv.STRIPE_METER_TOOL_CALLS_ID,
          generation: typedEnv.STRIPE_METER_GENERATIONS_ID,
        };

        const yesterday = new Date(Date.UTC(
          new Date().getUTCFullYear(), new Date().getUTCMonth(), new Date().getUTCDate() - 1,
        ));
        const today = new Date(Date.UTC(
          new Date().getUTCFullYear(), new Date().getUTCMonth(), new Date().getUTCDate(),
        ));

        // For each (tenant, event_type) seen yesterday in TimescaleDB usage_hourly:
        const rows = await step.run('list-yesterday-tenant-event', async () => {
          const result = await db.execute(sql`
            SELECT
              p.org_id          AS tenant_id,
              uh.event_type     AS event_type,
              SUM(uh.call_count)::int AS timescale_count
            FROM usage_hourly uh
            JOIN deployments d ON d.id = uh.deployment_id
            JOIN generations g ON g.id = d.generation_id
            JOIN projects p    ON p.id = g.project_id
            WHERE uh.bucket >= ${yesterday}
              AND uh.bucket <  ${today}
            GROUP BY p.org_id, uh.event_type
          `);
          return result.rows as Array<{ tenant_id: string; event_type: string; timescale_count: number }>;
        });

        for (const row of rows) {
          const meterId = METER_ID_BY_EVENT[row.event_type];
          if (!meterId) continue;

          await step.run(`reconcile-${row.tenant_id}-${row.event_type}`, async () => {
            const org = await db.query.organizations.findFirst({ where: (o, { eq }) => eq(o.id, row.tenant_id) });
            if (!org?.stripe_customer_id) return;

            const stripeRes = await stripe.billing.meters.eventSummaries.list({
              id: meterId,
              customer: org.stripe_customer_id,
              start_time: Math.floor(yesterday.getTime() / 1000),
              end_time: Math.floor(today.getTime() / 1000),
            } as never);
            const stripeCount = stripeRes.data.reduce((acc: number, e: { aggregated_value: number }) => acc + (e.aggregated_value ?? 0), 0);

            const driftPct = Math.abs(row.timescale_count - stripeCount) / Math.max(row.timescale_count, 1) * 100;
            const alerted = driftPct > RECONCILIATION_DRIFT_THRESHOLD_PCT;

            try {
              await db.insert(reconciliation_log).values({
                id: ulid(),
                reconciliation_date: yesterday.toISOString().split('T')[0],
                event_type: row.event_type,
                tenant_id: row.tenant_id,
                timescale_count: row.timescale_count,
                stripe_count: stripeCount,
                drift_pct: String(driftPct.toFixed(3)),
                alerted,
              });
            } catch (err) {
              if ((err as { code?: string }).code === '23505') return;  // already reconciled this date+event+tenant
              throw err;
            }

            if (alerted) {
              await sendReconciliationAlert(typedEnv, driftPct, row.event_type, row.tenant_id);
            }
          });
        }

        return { reconciled: rows.length };
      },
    );

    import { Hono } from 'hono';
    import { zValidator } from '@hono/zod-validator';
    import { ulid } from 'ulid';
    import { eq } from 'drizzle-orm';
    import {
      DeploymentDriftPatchRequest,
      drift_events,
      deployments,
      generations,
    } from '@mcpgen/contracts';
    import { db } from '../../db.js';
    import type { AuthContext } from '../../middleware/auth.js';

    export const driftRoute = new Hono<{ Variables: { auth: AuthContext } }>();

    async function verifyDeploymentOwnership(deploymentId: string, orgId: string): Promise<boolean> {
      const dep = await db.query.deployments.findFirst({
        where: eq(deployments.id, deploymentId),
        with: { generation: { with: { project: { with: { organization: true } } } } },
      });
      return dep?.generation.project.organization.id === orgId;
    }

    driftRoute.get('/deployments/:id/drift-events', async (c) => {
      const auth = c.var.auth;
      if (auth.isM2M) return c.json({ error: 'forbidden', reason: 'm2m_cannot_read_drift' }, 403);
      if (!auth.organizationId) return c.json({ error: 'no_org_context' }, 400);
      const deploymentId = c.req.param('id');
      if (!await verifyDeploymentOwnership(deploymentId, auth.organizationId)) {
        return c.json({ error: 'not_found' }, 404);  // defense-in-depth: do not confirm existence
      }
      const events = await db.query.drift_events.findMany({
        where: eq(drift_events.deployment_id, deploymentId),
        orderBy: (de, { desc }) => [desc(de.detected_at)],
      });
      return c.json({ drift_events: events });
    });

    driftRoute.post('/drift-events/:id/regenerate', async (c) => {
      const auth = c.var.auth;
      if (auth.isM2M) return c.json({ error: 'forbidden', reason: 'm2m_cannot_regenerate' }, 403);
      if (!auth.organizationId) return c.json({ error: 'no_org_context' }, 400);
      const driftEventId = c.req.param('id');
      const event = await db.query.drift_events.findFirst({
        where: eq(drift_events.id, driftEventId),
        with: { deployment: { with: { generation: { with: { project: { with: { organization: true } } } } } } },
      });
      if (!event || event.deployment.generation.project.organization.id !== auth.organizationId) {
        return c.json({ error: 'not_found' }, 404);
      }
      const newGenId = ulid();
      await db.insert(generations).values({
        id: newGenId,
        spec_id: event.deployment.generation.spec_id,
        project_id: event.deployment.generation.project_id,
        triggered_by: 'drift_manual',
        status: 'queued',
      });
      await db.update(drift_events).set({ status: 'regenerating', resolved_by_generation_id: newGenId }).where(eq(drift_events.id, driftEventId));
      return c.json({ job_id: newGenId, sse_url: `/api/v1/jobs/${newGenId}/stream` });
    });

    driftRoute.patch(
      '/deployments/:id',
      zValidator('json', DeploymentDriftPatchRequest),
      async (c) => {
        const auth = c.var.auth;
        if (auth.isM2M) return c.json({ error: 'forbidden', reason: 'm2m_cannot_patch_deployment' }, 403);
        if (!auth.organizationId) return c.json({ error: 'no_org_context' }, 400);
        const deploymentId = c.req.param('id');
        if (!await verifyDeploymentOwnership(deploymentId, auth.organizationId)) {
          return c.json({ error: 'not_found' }, 404);
        }
        const body = c.req.valid('json');
        await db.update(deployments).set({ auto_regenerate_on_drift: body.auto_regenerate_on_drift }).where(eq(deployments.id, deploymentId));
        return c.json({ ok: true, auto_regenerate_on_drift: body.auto_regenerate_on_drift });
      },
    );
