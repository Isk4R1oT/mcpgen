    4. `.pre-commit-hooks/no-fourth-namespace.sh` (D-08, defends T-1-05):
       ```bash
       #!/usr/bin/env bash
       # D-08 / Pitfall #11: never create a 4th CF dispatch namespace.
       # Allowed: mcpgen-prod, mcpgen-staging, mcpgen-sandbox.
       set -euo pipefail
       ALLOWED='^(mcpgen-prod|mcpgen-staging|mcpgen-sandbox)$'
       # Search for any wrangler dispatch-namespace create invocations:
       declare -a HITS=()
       while IFS= read -r line; do
         HITS+=("$line")
       done < <(git grep -h -E "wrangler\s+dispatch-namespace\s+create\s+\S+" \
         -- 'infrastructure/cloudflare/**' '**/*.toml' '**/*.sh' '**/*.md' 2>/dev/null \
         | sed -E "s/.*wrangler\s+dispatch-namespace\s+create\s+([a-z0-9-]+).*/\1/" \
         | sort -u || true)
       UNIQUE_COUNT=${#HITS[@]}
       FAILED=0
       for ns in "${HITS[@]:-}"; do
         if [[ -n "$ns" && ! "$ns" =~ $ALLOWED ]]; then
           echo "ERROR: forbidden CF dispatch namespace '$ns' detected. Allowed: mcpgen-prod, mcpgen-staging, mcpgen-sandbox (D-08, Pitfall #11)."
           FAILED=1
         fi
       done
       if [[ "$UNIQUE_COUNT" -gt 3 ]]; then
         echo "ERROR: $UNIQUE_COUNT distinct CF dispatch namespaces found in tracked files; max 3 allowed (D-08)."
         FAILED=1
       fi
       exit "$FAILED"
       ```

       **Date:** 2026-04-26
       **Status:** Accepted (Phase 1)
       **Decision drivers:** D-21, OPS-02, Pitfall #26

       - Workstream isolation is real, not aspirational.
