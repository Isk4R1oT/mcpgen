       ```bash
       #!/usr/bin/env bash
       # D-17: Hyperdrive in front of Neon for BFF (apps/api) + dispatch worker (apps/dispatch).
       # Engine connects directly to Neon (Fly Machines have no edge constraint).
       set -euo pipefail

       if [[ -z "${DATABASE_URL:-}" ]]; then
         echo "ERROR: DATABASE_URL must be set (Neon connection string with sslmode=require)."
         echo "       Source from .env.local: set -a && source .env.local && set +a"
         exit 1
       fi

       NAME="mcpgen-pg"

       echo "Checking for existing Hyperdrive '$NAME'..."
       EXISTING=$(wrangler hyperdrive list 2>/dev/null | awk -v n="$NAME" '$0 ~ n {print $1; exit}' || true)

       if [[ -n "$EXISTING" ]]; then
         echo "OK: Hyperdrive '$NAME' already exists (ID: $EXISTING) — idempotent skip."
         HYPERDRIVE_ID="$EXISTING"
       else
         echo "Creating Hyperdrive '$NAME'..."
         OUTPUT=$(wrangler hyperdrive create "$NAME" --connection-string "$DATABASE_URL" 2>&1)
         echo "$OUTPUT"
         HYPERDRIVE_ID=$(echo "$OUTPUT" | grep -oE 'id = "[a-f0-9-]+"' | sed 's/id = "//; s/"//')
         if [[ -z "$HYPERDRIVE_ID" ]]; then
           echo "ERROR: failed to extract Hyperdrive ID from create output"
           exit 1
         fi
         echo "Created Hyperdrive ID: $HYPERDRIVE_ID"
       fi

       echo ""
       echo "Updating wrangler.toml bindings..."
       for toml in apps/api/wrangler.toml apps/dispatch/wrangler.toml; do
         if [[ -f "$toml" ]]; then
           # Cross-platform sed (BSD on macOS uses different -i syntax than GNU)
           sed -i.bak "s|REPLACE_WITH_HYPERDRIVE_ID|$HYPERDRIVE_ID|g" "$toml"
           rm -f "$toml.bak"
           echo "Updated: $toml"
         fi
       done

       echo ""
       echo "OK: Hyperdrive '$NAME' bound in apps/api + apps/dispatch."
       echo "Verify: grep -A2 'binding = \"HYPERDRIVE\"' apps/api/wrangler.toml apps/dispatch/wrangler.toml"
       ```
