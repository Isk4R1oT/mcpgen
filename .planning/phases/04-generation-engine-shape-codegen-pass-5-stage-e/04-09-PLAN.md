      from mcpgen_engine.stages.stage_e.scaffold import GeneratedFile
      from mcpgen_engine.stages.stage_e.schemas import _render_inputs_hash
      from mcpgen_engine.stages.stage_e.template_loader import ENVIRONMENT

      _log = structlog.get_logger(__name__)

      AuthMode = Literal["passthrough", "stored", "oauth"]


      def select_auth_mode(pass_0_output: Pass0Output) -> AuthMode:
          """Per Phase 2 D-22 mapping: oauth2→oauth, aws_signature→stored, default passthrough."""
          for reqs in pass_0_output.auth_requirements.values():
              for req in reqs:
                  scheme = getattr(req, "scheme", None) or getattr(req, "type", None)
                  if scheme == "oauth2":
                      return "oauth"
          for reqs in pass_0_output.auth_requirements.values():
              for req in reqs:
                  scheme = getattr(req, "scheme", None) or getattr(req, "type", None)
                  if scheme == "aws_signature":
                      return "stored"
          return "passthrough"


      def render_auth_files(
          auth_mode: AuthMode,
          pass_0_output: Pass0Output,
      ) -> list[GeneratedFile]:
          """Render src/auth/middleware.ts + src/auth/credentials.ts."""
          context: dict[str, Any] = {
              "auth_mode": auth_mode,
          }
          out: list[GeneratedFile] = []
          for relative_path, template_name in (
              ("src/auth/middleware.ts", "auth_middleware.ts.j2"),
              ("src/auth/credentials.ts", "auth_credentials.ts.j2"),
          ):
              rendered = ENVIRONMENT.get_template(template_name).render(**context)
              out.append(GeneratedFile(
                  relative_path=relative_path,
                  content=rendered,
                  render_template=template_name,
                  render_inputs_hash=_render_inputs_hash(context),
              ))
          _log.info("stage_e.auth.render_complete", auth_mode=auth_mode)
          return out
