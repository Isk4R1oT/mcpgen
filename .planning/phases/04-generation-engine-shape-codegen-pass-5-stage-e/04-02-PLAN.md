      """Pass 5 Phase 2 — deterministic outputSchema extraction.

      Per CONTEXT D-05 Phase 2: walks `RawIR.endpoints[*].responses[200].schema`
      and produces per-tool JSON Schema dict with the canonical Pass 5 wrapper
      `{id, object_type, data, metadata}` (universal-tool branch table per
      Pass 5 design §1.6).

      Universal `list_*` tools that subsume multiple collections produce a
      `oneOf` aggregated schema. Conservative-format Zod fallback (Pitfall
      #33) lives in plan 04-07's Stage E `outputs.ts.j2` template — this
      module emits raw JSON Schema.

      Threats addressed:
      - T-04-02-spec-leak: structural-only logging; raw spec content NEVER
        logged.
      - T-04-02-IR-mutation: pure functions; outputs are new dicts (deep-
        copy of inner spec).
      - T-04-02-low-confidence-silent: when spec response is
        `additionalProperties: true` or empty, set `inference_low_confidence
        =True` so plan 04-05 can surface a warning in Pass5Output flags.

      References:
      - 04-CONTEXT.md D-05 Phase 2
      - 04-RESEARCH.md §"Component Responsibilities" output_schema.py row
      - docs/mcpgen-pass-5-design.md §1.1 + §1.6
      """

      from __future__ import annotations

      import copy
      from collections import Counter
      from typing import Any, Final

      import structlog
      from mcpgen_ir.types import (
          Endpoint,
          Pass1Output,
          RawIR,
          ToolTaxonomyEntry,
          Type,
      )
      from pydantic import BaseModel, ConfigDict

      _log = structlog.get_logger(__name__)


      class OutputSchemaSpec(BaseModel):
          """Internal Pass 5 type — NOT exported in IR.

          Consumed by:
          - field_ranking.py (plan 04-03) — input for LLM importance ranking.
          - validation.py (plan 04-05) — cross-tool consistency checks.
          - final_assembly.py (plan 04-05) — produces FinalTool.outputSchema.
          """

          model_config = ConfigDict(extra="forbid")

          tool_name: str
          # Per-field JSON Schema (just the inner shape; metadata wrapper added
          # at to_json_schema() time).
          fields: dict[str, dict[str, Any]]
          source_endpoint_id: str
          wrapping_object_type: str
          # True when spec response is `additionalProperties: true` OR has empty
          # properties dict — surfaced as a warning by plan 04-05.
          inference_low_confidence: bool = False


      def wrap_with_metadata(
          inner_data_schema: dict[str, Any],
          object_type: str,
      ) -> dict[str, Any]:
          """Canonical Pass 5 outputSchema wrapper (D-05 + Pass 5 design §1.1)."""
          return {
              "type": "object",
              "properties": {
                  "id": {"type": "string", "description": "Smart ID"},
                  "object_type": {"type": "string", "const": object_type},
                  "data": copy.deepcopy(inner_data_schema),
                  "metadata": {
                      "type": "object",
                      "properties": {
                          "fetched_at": {"type": "string", "format": "date-time"},
                          "source_endpoint": {"type": "string"},
                      },
                      "required": ["fetched_at", "source_endpoint"],
                  },
              },
              "required": ["id", "object_type", "data"],
              "additionalProperties": False,
          }


      def _build_search_envelope(
          inner: dict[str, Any], collections: list[str],
      ) -> dict[str, Any]:
          """Pass 5 design §1.6: search → {results: array<{id, score?, preview}>, total_count?, next_cursor?}."""
          return {
              "type": "object",
              "properties": {
                  "results": {
                      "type": "array",
                      "items": {
                          "type": "object",
                          "properties": {
                              "id": {"type": "string", "description": "Smart ID"},
                              "score": {"type": "number"},
                              "preview": {"type": "object", "additionalProperties": True},
                          },
                          "required": ["id"],
                      },
                  },
                  "total_count": {"type": "integer", "minimum": 0},
                  "next_cursor": {"type": "string"},
              },
              "required": ["results"],
          }


      def _build_fetch_envelope(
          inner: dict[str, Any], collections: list[str],
      ) -> dict[str, Any]:
          """fetch → single object passes through to data field."""
          return inner if inner else {"type": "object", "additionalProperties": True}


      def _build_list_objects_envelope(
          inner: dict[str, Any], collections: list[str],
      ) -> dict[str, Any]:
          """list_objects → {items: array<oneOf-aggregated>, next_cursor?, total_count?}."""
          # When subsuming multiple collections, build oneOf array.
          item_schema: dict[str, Any]
          if len(collections) > 1 and isinstance(inner, dict) and "oneOf" in inner:
              item_schema = inner
          elif inner:
              item_schema = inner
          else:
              item_schema = {"type": "object", "additionalProperties": True}
          return {
              "type": "object",
              "properties": {
                  "items": {"type": "array", "items": item_schema},
                  "next_cursor": {"type": "string"},
                  "offset": {"type": "integer"},
                  "total_count": {"type": "integer"},
              },
              "required": ["items"],
          }


      def _build_list_collections_envelope(
          inner: dict[str, Any], collections: list[str],
      ) -> dict[str, Any]:
          """list_collections → {items: array<{name, schema?, ...}>}."""
          return {
              "type": "object",
              "properties": {
                  "items": {
                      "type": "array",
                      "items": {
                          "type": "object",
                          "properties": {
                              "name": {"type": "string"},
                              "description": {"type": "string"},
                              "schema": {"type": "object", "additionalProperties": True},
                          },
                          "required": ["name"],
                      },
                  },
              },
              "required": ["items"],
          }


      def _build_upsert_envelope(
          inner: dict[str, Any], collections: list[str],
      ) -> dict[str, Any]:
          """upsert → returns the upserted object (single)."""
          return inner if inner else {"type": "object", "additionalProperties": True}


      def _build_delete_envelope(
          inner: dict[str, Any], collections: list[str],
      ) -> dict[str, Any]:
          """delete → {deleted_count: integer, success: boolean}."""
          return {
              "type": "object",
              "properties": {
                  "deleted_count": {"type": "integer", "minimum": 0},
                  "success": {"type": "boolean"},
              },
              "required": ["deleted_count", "success"],
          }


      _UNIVERSAL_ENVELOPE_BUILDERS: Final[dict[str, Any]] = {
          "search": _build_search_envelope,
          "fetch": _build_fetch_envelope,
          "list_objects": _build_list_objects_envelope,
          "list_collections": _build_list_collections_envelope,
          "upsert": _build_upsert_envelope,
          "delete": _build_delete_envelope,
      }


      def _resolve_response_schema(endpoint: Endpoint) -> dict[str, Any]:
          """Pull response schema with 200 → 201 → first 2xx fallback."""
          for code in ("200", "201", "202", "204"):
              response = endpoint.responses.get(code)
              if response is not None and getattr(response, "schema_", None):
                  schema = response.schema_
                  if isinstance(schema, dict):
                      return schema
          # Try any 2xx as last resort.
          for code, response in endpoint.responses.items():
              if code.startswith("2") and getattr(response, "schema_", None):
                  if isinstance(response.schema_, dict):
                      return response.schema_
          return {}


      def extract_output_schema_for_tool(
          tool: ToolTaxonomyEntry,
          raw_ir: RawIR,
      ) -> OutputSchemaSpec:
          """Per-tool deterministic outputSchema spec extractor."""
          endpoints_by_id: dict[str, Endpoint] = {
              f"{e.method.value} {e.path}": e for e in raw_ir.endpoints
          }
          if not tool.source_endpoints:
              return OutputSchemaSpec(
                  tool_name=tool.name,
                  fields={},
                  source_endpoint_id="",
                  wrapping_object_type=tool.name,
                  inference_low_confidence=True,
              )
          first_endpoint_id = tool.source_endpoints[0]
          endpoint = endpoints_by_id.get(first_endpoint_id)
          if endpoint is None:
              return OutputSchemaSpec(
                  tool_name=tool.name,
                  fields={},
                  source_endpoint_id=first_endpoint_id,
                  wrapping_object_type=tool.name,
                  inference_low_confidence=True,
              )

          # Universal upsert prefers 201; others prefer 200.
          if tool.type == Type.universal and tool.universal_tool == "upsert":
              raw_schema: dict[str, Any] = {}
              for code in ("201", "200", "202"):
                  resp = endpoint.responses.get(code)
                  if resp is not None and getattr(resp, "schema_", None):
                      if isinstance(resp.schema_, dict):
                          raw_schema = resp.schema_
                          break
          else:
              raw_schema = _resolve_response_schema(endpoint)

          # Detect low-confidence (additionalProperties: true OR empty properties).
          low_confidence = (
              not raw_schema
              or raw_schema.get("additionalProperties") is True
              or not raw_schema.get("properties")
          )

          # Aggregate across collections for list_objects (oneOf) when needed.
          collections: list[str] = []
          if isinstance(tool.routing, dict):
              collections = list(tool.routing.get("collections", []) or [])

          # Build envelope per universal-tool branch.
          inner = raw_schema
          if (
              tool.type == Type.universal
              and tool.universal_tool == "list_objects"
              and len(collections) > 1
          ):
              # oneOf aggregation across collections — collect each collection's schema.
              one_of_items: list[dict[str, Any]] = []
              # NOTE: real collection-schema lookup happens via raw_ir if endpoints
              # carry per-collection schemas; for tools where this isn't available
              # fall back to single-schema with additionalProperties: true.
              one_of_items.append(raw_schema if raw_schema else {"type": "object", "additionalProperties": True})
              inner = {"oneOf": one_of_items, "additionalProperties": True}

          if tool.type == Type.universal and tool.universal_tool in _UNIVERSAL_ENVELOPE_BUILDERS:
              builder = _UNIVERSAL_ENVELOPE_BUILDERS[tool.universal_tool]
              data_schema = builder(inner, collections)
          else:
              data_schema = inner if inner else {"type": "object", "additionalProperties": True}

          # Determine wrapping object_type.
          object_type = (
              collections[0] if len(collections) == 1
              else tool.universal_tool or tool.name
          )

          # Convert data_schema into per-field properties dict for OutputSchemaSpec.
          if data_schema.get("properties"):
              fields = {k: copy.deepcopy(v) for k, v in data_schema["properties"].items()}
          else:
              fields = {}

          return OutputSchemaSpec(
              tool_name=tool.name,
              fields=fields,
              source_endpoint_id=first_endpoint_id,
              wrapping_object_type=str(object_type),
              inference_low_confidence=bool(low_confidence),
          )


      def extract_all_output_schemas(
          pass_1_output: Pass1Output,
          raw_ir: RawIR,
      ) -> dict[str, OutputSchemaSpec]:
          """Walk every tool and extract its outputSchema spec."""
          result: dict[str, OutputSchemaSpec] = {}
          for tool in pass_1_output.tools:
              result[tool.name] = extract_output_schema_for_tool(tool, raw_ir)

          schema_kinds = Counter(
              s.wrapping_object_type for s in result.values()
          )
          low_confidence_count = sum(
              1 for s in result.values() if s.inference_low_confidence
          )
          _log.info(
              "pass_5.output_schema.extract_complete",
              tool_count=len(result),
              low_confidence_count=low_confidence_count,
              distinct_object_types=len(schema_kinds),
          )
          return result

          """Convert OutputSchemaSpec to the wrapped JSON Schema (FinalTool.outputSchema value)."""
          inner_data_schema: dict[str, Any] = (

              if spec.fields
              else {"type": "object", "additionalProperties": True}
          )
          return wrap_with_metadata(inner_data_schema, spec.wrapping_object_type)
