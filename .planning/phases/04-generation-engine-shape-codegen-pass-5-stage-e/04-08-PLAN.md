      export class UpstreamError extends Error {
        public readonly status: number;
        public readonly body: string;
        public readonly retry_after?: number;
        constructor(status: number, body: string, retry_after?: number) {
          super(`Upstream ${status}: ${body.slice(0, 200)}`);
          this.status = status;
          this.body = body;
          this.retry_after = retry_after;
        }
      }

      export interface UpstreamRequestOptions {
        readonly method: string;
        readonly path: string;
        readonly ctx: { readonly upstreamCredential?: string };
        readonly body?: unknown;
      }

      export async function upstreamRequest(opts: UpstreamRequestOptions, baseUrl: string): Promise<unknown> {
        for (let attempt = 0; attempt <= MAX_RETRIES; attempt++) {
          const headers: Record<string, string> = { "content-type": "application/json" };
          if (opts.ctx.upstreamCredential) {
            headers["authorization"] = opts.ctx.upstreamCredential.startsWith("Bearer ")
              ? opts.ctx.upstreamCredential
              : `Bearer ${opts.ctx.upstreamCredential}`;
          }
          const resp = await fetch(`${baseUrl}${opts.path}`, {
            method: opts.method,
            headers,
            body: opts.body ? JSON.stringify(opts.body) : undefined,
          });
          if (resp.status >= 500 && attempt < MAX_RETRIES) {
            await new Promise((r) => setTimeout(r, BACKOFF_MS_BASE * Math.pow(2, attempt)));
            continue;
          }
          if (resp.ok) return await resp.json();
          const body = await resp.text();
          const retry_after_h = resp.headers.get("retry-after");
          throw new UpstreamError(resp.status, body, retry_after_h ? Number(retry_after_h) : undefined);
        }
        throw new UpstreamError(0, "All retries exhausted");
      }

      export function applyFieldFilter(
        data: unknown,
        alwaysInclude: readonly string[],
        optInRequested: readonly string[],
      ): unknown {
        if (!data || typeof data !== "object") return data;
        const obj = data as Record<string, unknown>;
        const allowed = new Set([...alwaysInclude, ...optInRequested]);
        // Empty alwaysInclude = pass-through.
        if (alwaysInclude.length === 0 && optInRequested.length === 0) return obj;
        const out: Record<string, unknown> = {};
        for (const k of Object.keys(obj)) {
          if (allowed.has(k) || allowed.has("*")) out[k] = obj[k];
        }
        return out;
      }
