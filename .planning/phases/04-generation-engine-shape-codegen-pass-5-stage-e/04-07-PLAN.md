      _log = structlog.get_logger(__name__)


      def _render_inputs_hash(context: dict[str, Any]) -> str:
          canonical = json.dumps(context, sort_keys=True, default=str)
          return hashlib.sha256(canonical.encode("utf-8")).hexdigest()


      def render_schema_files(
          final_tools: list[FinalTool],
          pass_1_output: Pass1Output,
      ) -> list[GeneratedFile]:
          """Render the 3 schema files."""
          tool_payloads = [
              {
                  "name": t.name,
                  "input_schema": t.inputSchema,
                  "output_schema": t.outputSchema,
                  "type": t.type.value,
                  "universal_tool": t.universal_tool,
                  "source_endpoints": list(t.source_endpoints),
                  "routing": t.routing,
              }
              for t in final_tools
          ]
          context_inputs = {"final_tools": tool_payloads}
          context_outputs = {"final_tools": tool_payloads}
          context_routing = {
              "final_tools": tool_payloads,
              "pass_1_routing": pass_1_output.routing.model_dump() if hasattr(pass_1_output, "routing") and pass_1_output.routing else {},

          return [
              GeneratedFile(
                  relative_path="src/schemas/inputs.ts",

              ),

      // Generated by mcpgen — Stage E Phase 2 (schemas/outputs.ts).
      // Pitfall #33: dual export. richSchemas = Zod 4 native (with format strings).
      // default = conservative (no format strings — older client safe).
      import { z } from "zod";
      import type { JsonSchema } from "@modelcontextprotocol/sdk/types.js";

      type AnyJson = Record<string, unknown>;

      function stripFormat(node: AnyJson): AnyJson {
        if (typeof node !== "object" || node === null) return node;
        const out: AnyJson = Array.isArray(node) ? [] : {};
        for (const [k, v] of Object.entries(node)) {
          if (k === "format" || k === "pattern") continue;
          if (typeof v === "object" && v !== null) {
            (out as Record<string, unknown>)[k] = stripFormat(v as AnyJson);
          } else {
            (out as Record<string, unknown>)[k] = v;
          }
        }
        return out;
      }

      {% for tool in final_tools -%}
      // {{ tool.name }} richSchema (Zod-derived; includes format strings).
      const _richOutputSchema_{{ tool.name }}: AnyJson = {{ tool.output_schema | tojson }};
