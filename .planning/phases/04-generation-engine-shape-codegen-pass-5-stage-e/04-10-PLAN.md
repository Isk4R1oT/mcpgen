      ```python
      """Stage E Phase 5 — per-tool-type handler renderer (D-31)."""
      from typing import Any, Final
      import structlog

      from mcpgen_ir.types import FinalTool, Pass1Output, Type
      from mcpgen_engine.stages.stage_e.scaffold import GeneratedFile
      from mcpgen_engine.stages.stage_e.schemas import _render_inputs_hash
      from mcpgen_engine.stages.stage_e.template_loader import ENVIRONMENT

      _log = structlog.get_logger(__name__)

      # D-31 dispatch table — (tool.type.value, tool.universal_tool) → template name.
      _TEMPLATE_BY_TOOL_TYPE: Final[dict[tuple[str, str | None], str]] = {
          ("universal", "search"): "tool_search.ts.j2",
          ("universal", "fetch"): "tool_fetch.ts.j2",
          ("universal", "list_collections"): "tool_list_collections.ts.j2",
          ("universal", "list_objects"): "tool_list_objects.ts.j2",
          ("universal", "upsert"): "tool_upsert.ts.j2",
          ("universal", "delete"): "tool_delete.ts.j2",
          ("action", None): "tool_action.ts.j2",
          ("workflow", None): "tool_workflow.ts.j2",
          ("specialized", None): "tool_specialized.ts.j2",
      }


      def _select_template(tool: FinalTool) -> str:
          tool_type_value = tool.type.value if hasattr(tool.type, "value") else str(tool.type)
          if tool_type_value == "universal":
              key = (tool_type_value, tool.universal_tool)
          else:
              key = (tool_type_value, None)
          if key not in _TEMPLATE_BY_TOOL_TYPE:
              raise KeyError(f"No handler template for tool_type={key!r}")
          return _TEMPLATE_BY_TOOL_TYPE[key]


      def _to_pascal(name: str) -> str:
          return "".join(part.capitalize() for part in name.split("_") if part)


      def render_tool_handler(tool: FinalTool, pass_1_output: Pass1Output) -> tuple[str, str, str]:
          """Returns (relative_path, content, template_name)."""
          template_name = _select_template(tool)
          context: dict[str, Any] = {
              "tool": {
                  "name": tool.name,
                  "type": tool.type.value if hasattr(tool.type, "value") else str(tool.type),
                  "universal_tool": tool.universal_tool,
                  "title": tool.title,
                  "annotations": tool.annotations.model_dump() if hasattr(tool.annotations, "model_dump") else tool.annotations,
                  "input_schema": tool.inputSchema,
                  "output_schema": tool.outputSchema,
                  "description_text": tool.description.purpose if hasattr(tool.description, "purpose") else "",
                  "source_endpoints": list(tool.source_endpoints),
                  "routing": tool.routing,
                  "response_config": tool.response_config.model_dump() if hasattr(tool.response_config, "model_dump") else tool.response_config,
              },
              "register_function_name": f"register{_to_pascal(tool.name)}",
          }
          rendered = ENVIRONMENT.get_template(template_name).render(**context)
          return f"src/tools/{tool.name}.ts", rendered, template_name


      def render_all_tool_handlers(
          final_tools: list[FinalTool], pass_1_output: Pass1Output,
      ) -> list[GeneratedFile]:
          """Walk tools + emit one TS file per tool + tools/index.ts registry."""
          out: list[GeneratedFile] = []
          for tool in final_tools:
              relative_path, content, template_name = render_tool_handler(tool, pass_1_output)
              out.append(GeneratedFile(
                  relative_path=relative_path,
                  content=content,
                  render_template=template_name,
                  render_inputs_hash=_render_inputs_hash({"tool_name": tool.name}),
              ))
          # tools/index.ts registry.
          index_context = {
              "tools": [
                  {"name": t.name, "register_function_name": f"register{_to_pascal(t.name)}"}
                  for t in final_tools
              ],
          }
          out.append(GeneratedFile(
              relative_path="src/tools/index.ts",
              content=ENVIRONMENT.get_template("tools_index.ts.j2").render(**index_context),
              render_template="tools_index.ts.j2",
              render_inputs_hash=_render_inputs_hash(index_context),
          ))
          _log.info("stage_e.tools.render_complete", tool_count=len(final_tools), file_count=len(out))
          return out
      ```

      ```jinja
      // Tool type: universal/fetch

      import type { McpServer } from "@modelcontextprotocol/sdk/server/mcp.js";
      import { parseSmartId, makeSmartId } from "../runtime/smart_id.js";
      import { upstreamRequest } from "../runtime/upstream.js";
      import { applyFieldFilter, assembleStructuredContent } from "../runtime/response_shaping.js";
      import { applyTruncation } from "../runtime/truncation.js";
      import { handleUpstreamError } from "../runtime/errors.js";
      import { routing } from "../schemas/routing.js";
      import { SERVER_NAME_TEMPLATE, UPSTREAM_BASE_URL } from "../config.js";

      export function {{ register_function_name }}(server: McpServer): void {
        server.tool(
          "{{ tool.name }}",
          {{ tool.description_text | tojson }},
          z.object({
            id: z.string().describe("Smart ID. Format: {server}:object:{collection}:{identifier}"),
            properties: z.array(z.string()).optional().describe("Subset of fields to include (use ['*'] for all)."),
          }).shape,
          async (args, extra) => {
            try {
              const ctx = (extra as { _ctx?: { clientVersion?: string; upstreamCredential?: string } })._ctx ?? {};
              const parts = parseSmartId(args.id);
              const routeKey = `{{ tool.name }}:` + parts.collection;
              const route = routing[routeKey];
              if (!route) {
                return {
                  isError: true,
                  content: [{ type: "text" as const, text: `Unknown collection: ${parts.collection}. Use list_collections() first.` }],
                };
              }
              const upstreamData = await upstreamRequest(
                { method: route.method, path: route.path.replace("{id}", parts.identifier), ctx },
                UPSTREAM_BASE_URL,
              );
              const filtered = applyFieldFilter(upstreamData, [], args.properties ?? []);
              const truncated = applyTruncation("fetch", filtered, { N: 0, Total: 0 });
              return assembleStructuredContent(truncated.data, ctx.clientVersion);
            } catch (err) {
              return handleUpstreamError(err, "{{ tool.name }}");
            }
          },
          {
            title: {{ tool.title | tojson }},
            annotations: {{ tool.annotations | tojson }},
          },
        );
      ```
