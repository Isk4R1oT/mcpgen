      """

      from __future__ import annotations

      from typing import Any, Final

      import structlog
      from mcpgen_ir.types import Pass1Output, ToolTaxonomyEntry, Type
      from pydantic import BaseModel, ConfigDict

      from mcpgen_engine.passes.pass_5.templates import _TRUNCATION_TEMPLATES

      _log = structlog.get_logger(__name__)

      TOOL_TYPE_KEYS: Final[frozenset[str]] = frozenset(
          {"search", "list_objects", "list_collections", "fetch", "upsert", "delete", "action", "workflow"}

      class TruncationConfig1(BaseModel):
          """Internal Pass 5 type — NOT exported in IR.

          The IR-exported `TruncationConfig` type lives in
          `packages/ir/python/types.py` and is shaped slightly differently
          (per-tool runtime config). Plan 04-05 final_assembly converts
          `TruncationConfig1` → IR `TruncationConfig` shape.
          """

          model_config = ConfigDict(extra="forbid")
          tool_type: str
          threshold_tokens: int
          template: str


      def resolve_tool_type(tool: ToolTaxonomyEntry) -> str:
          """Map ToolTaxonomyEntry → truncation-template key."""
          if tool.type == Type.universal and tool.universal_tool in TOOL_TYPE_KEYS:
              return tool.universal_tool
          if tool.type == Type.action:
              return "action"
          if tool.type == Type.workflow:
              return "workflow"
          if tool.type == Type.specialized:
              # Specialized reads route to `fetch` analog (single-object semantics).
              return "fetch"
          # Defensive default — should be unreachable given Type enum exhaustion.
          return "fetch"


      def build_truncation_config(
          tool: ToolTaxonomyEntry,
          pagination_strategy_value: str,
      ) -> TruncationConfig1:
          """Combine tool-type lookup with the frozen template table."""
          tool_type = resolve_tool_type(tool)
          row = _TRUNCATION_TEMPLATES[tool_type]
          threshold = row["threshold_tokens"]
          template = row["template"]
          assert isinstance(threshold, int)
          assert isinstance(template, str)
          return TruncationConfig1(
              tool_type=tool_type,
              threshold_tokens=threshold,
              template=template,
          )


      def apply_truncation_template(
          tool_type: str,
          n_shown: int,
          total: int,
          **kwargs: Any,
      ) -> str:
          """Runtime substitution helper (Python-side reference; Stage E mirrors in TS).

          Raises KeyError if a required placeholder isn't provided in kwargs.
          """
          row = _TRUNCATION_TEMPLATES[tool_type]
          template = row["template"]
          assert isinstance(template, str)
          return template.format(
              N=n_shown,
              Total=total,
              Total_minus_N=max(0, total - n_shown),
              **kwargs,
          )


      def build_all_truncation_configs(
          pass_1_output: Pass1Output,
          pagination_strategy_value: str,
      ) -> dict[str, TruncationConfig1]:
          """Walk every tool and build its truncation config."""
          result: dict[str, TruncationConfig1] = {}
          for tool in pass_1_output.tools:
              result[tool.name] = build_truncation_config(tool, pagination_strategy_value)
          _log.info(
              "pass_5.truncation.build_complete",
              tool_count=len(result),
              tool_type_breakdown={
                  ttype: sum(1 for c in result.values() if c.tool_type == ttype)
                  for ttype in TOOL_TYPE_KEYS
              },
          )
          return result
