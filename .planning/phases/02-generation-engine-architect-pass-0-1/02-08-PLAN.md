**A. `cache/keys.py`** (RESEARCH lines 740-770 verbatim, with full L3 helper):

```python
from __future__ import annotations
import hashlib
import json
from importlib.metadata import version


def _engine_version() -> str:
    """Read engine semver from installed package metadata.

    `pyproject.toml [project] version = "0.0.0"` → installed as "0.0.0".
    Bumping the version invalidates ALL cache layers (D-40) by changing
    every key.
    """
    return version("mcpgen-generation-engine")


def _canonical_json_sha256(value: dict) -> str:
    canonical = json.dumps(value, sort_keys=True, separators=(",", ":"))
    return hashlib.sha256(canonical.encode("utf-8")).hexdigest()


def l1_key(spec_hash: str) -> str:
    """L1 cache key (D-37): spec-sha + engine_version."""
    raw = f"l1:{_engine_version()}:{spec_hash}"
    return hashlib.sha256(raw.encode("utf-8")).hexdigest()


def l2_key(
    *,
    pass_name: str,             # "pass_0" | "pass_1"
    pass_version: str,          # bumped manually when pass logic changes; e.g. "1"
    pass_input: dict,           # canonical input — for Pass 0 = serialized RawIR; for Pass 1 = Pass0Output
    sampling_profile_label: str,  # "PASS_0_SETTINGS" | "PASS_1_SETTINGS"
) -> str:
    """L2 cache key: pass_name + pass_version + sha(input) + model_id + sampling_profile."""
    input_hash = _canonical_json_sha256(pass_input)
    raw = (
        f"l2:{_engine_version()}:{pass_name}:{pass_version}:"
        f"qwen/qwen3-coder:{sampling_profile_label}:{input_hash}"
    )
    return hashlib.sha256(raw.encode("utf-8")).hexdigest()


def l3_key(
    *,
    tool_name: str,             # universal tool name OR action verb tool name
    tool_subset: dict,          # subset of input that's relevant to this single tool
    pass_version: str,
) -> str:
    """L3 cache key (Phase-3 relevance; Phase-2 ships infra only — D-37)."""
    subset_hash = _canonical_json_sha256(tool_subset)
    raw = f"l3:{_engine_version()}:{tool_name}:{pass_version}:{subset_hash}"
    return hashlib.sha256(raw.encode("utf-8")).hexdigest()
```

**B. `cache/l1.py`** (RESEARCH lines 772-809 verbatim — atomic writes, sharded layout, 30-day TTL):

```python
from __future__ import annotations
import gzip
import json
import os
import tempfile
import time
from pathlib import Path

from mcpgen_engine.settings import get_settings

CACHE_LAYER_NAME = "l1"
TTL_SECONDS = 30 * 86400  # D-40


def _cache_root() -> Path:
    return Path(get_settings().mcpgen_cache_dir)


def _path_for(layer: str, key: str) -> Path:
    """Sharded path: <root>/<layer>/<key[:2]>/<key[2:]>.json.gz.

    First 2 hex chars of sha256 → 256-way uniform shard, prevents single-directory
    listings >256 entries on filesystems with poor scaling (Pitfall: ext4 dir_index slowness).
    """
    return _cache_root() / layer / key[:2] / f"{key[2:]}.json.gz"


def get_l1(key: str) -> dict | None:
    p = _path_for(CACHE_LAYER_NAME, key)
    if not p.exists():
        return None
    if time.time() - p.stat().st_mtime > TTL_SECONDS:
        p.unlink(missing_ok=True)
        return None
    with gzip.open(p, "rt", encoding="utf-8") as f:
        return json.load(f)


def set_l1(key: str, value: dict) -> None:
    p = _path_for(CACHE_LAYER_NAME, key)
    p.parent.mkdir(parents=True, exist_ok=True, mode=0o700)
    # Atomic write: tempfile + rename (RESEARCH "Don't Hand-Roll" line 941)
    with tempfile.NamedTemporaryFile(
        mode="wt",
        encoding="utf-8",
        suffix=".tmp",
        dir=p.parent,
        delete=False,
    ) as tf:
        with gzip.open(tf.name, "wt", encoding="utf-8") as gzf:
            json.dump(value, gzf, sort_keys=True, separators=(",", ":"))
        tmp_path = Path(tf.name)
    tmp_path.replace(p)  # atomic on POSIX
    p.chmod(0o600)


def clear_layer() -> int:
    """Used by `mcpgen cache clear` (Phase 6+) — return count of files deleted.

    Phase-2 unused but exposed so test_cache_l1_l2.py::test_atomic_writes can clean up.
    """
    root = _cache_root() / CACHE_LAYER_NAME
    if not root.exists():
        return 0
    count = 0
    for p in root.rglob("*.json.gz"):
        p.unlink()
        count += 1
    return count
```

**A. `pipeline.py`** — orchestrator with SSE event emission:

```python
from __future__ import annotations
import asyncio
import re
from typing import AsyncIterator, Literal
from pydantic import BaseModel

from mcpgen_ir.types import RawIR, Pass0Output, Pass1Output
from mcpgen_engine.types import GenerationOptions  # define if not present
from mcpgen_engine.cache import get_l1, set_l1, l1_key
from mcpgen_engine.stages import stage_a
from mcpgen_engine.passes import pass_0, pass_1


# Mirrors Phase-1 frozen Zod contract (packages/contracts/src/generation-api.ts).
# `stage` enum union matches the Zod source.
class GenerationSseEvent(BaseModel):
    job_id: str                                                                # gen_<ULID>
    event_id: str                                                              # ULID monotonic per job
    stage: Literal["A", "B", "C", "D", "E", "F1", "F2", "F3", "completed", "failed"]
    status: Literal["started", "completed", "error"]
    partial_result: dict[str, str] | None = None
    error: dict[str, str] | None = None


class PipelineError(ValueError):
    """Wraps stage errors with stable code + user-facing message."""


def _new_event_id() -> str:
    """Monotonic ULID per call. Use python-ulid if pinned; else hand-roll."""
    from ulid import ULID  # `python-ulid` — verify pinned in pyproject; if not, install via `uv add python-ulid`
    return str(ULID())


def _event(job_id: str, stage: str, status: str, partial_result=None, error=None) -> GenerationSseEvent:
    return GenerationSseEvent(
        job_id=job_id, event_id=_new_event_id(),
        stage=stage, status=status,
        partial_result=partial_result, error=error,
    )


async def run_pipeline(
    *,
    spec_url: str | None,
    spec_content: str | None,
    options: GenerationOptions,
    job_id: str,
) -> AsyncIterator[GenerationSseEvent]:
    """Phase-2 pipeline: Stage A → Pass 0 → Pass 1 with L1 fast-path.

    Stage = retry boundary (engine v2 §5.1). Pass exceptions propagate as
    stage='failed' SSE events; do NOT retry inside the orchestrator (passes
    own their own retry logic per D-26 / D-34).
    """
    try:
        # Stage A
        yield _event(job_id, stage="A", status="started")
        raw_ir = await stage_a.run(spec_url=spec_url, spec_content=spec_content)

        # L1 fast-path — D-37 spec-sha lookup
        cache_key = l1_key(raw_ir.spec_hash)
        cached = get_l1(cache_key)
        if cached is not None:
            yield _event(job_id, stage="A", status="completed",
                         partial_result={"endpoint_count": str(len(raw_ir.endpoints)),
                                         "cache": "l1_hit"})
            yield _event(job_id, stage="completed", status="completed",
                         partial_result={"phase": "architect_complete", "cache": "l1_hit"})
            return

        yield _event(job_id, stage="A", status="completed",
                     partial_result={"endpoint_count": str(len(raw_ir.endpoints))})

        # Pass 0 + Pass 1 = Stage B (Architect)
        yield _event(job_id, stage="B", status="started",
                     partial_result={"phase": "pass_0"})
        pass_0_output = await pass_0.run(raw_ir, options)
        yield _event(job_id, stage="B", status="completed",
                     partial_result={"phase": "pass_0",
                                     "tool_plan_count": str(len(pass_0_output.tool_plans)),
                                     "dropped_count": str(len(pass_0_output.dropped_endpoints))})

        yield _event(job_id, stage="B", status="started",
                     partial_result={"phase": "pass_1"})
        pass_1_output = await pass_1.run(pass_0_output, raw_ir, options)
        yield _event(job_id, stage="B", status="completed",
                     partial_result={"phase": "pass_1",
                                     "final_tool_count": str(len(pass_1_output.tools)),
                                     "coverage_pct": str(pass_1_output.coverage_pct)})

        # Persist to L1 — full final IR for next-run zero-LLM second-run
        set_l1(cache_key, {
            "raw_ir": raw_ir.model_dump(mode="json"),
            "pass_0_output": pass_0_output.model_dump(mode="json"),
            "pass_1_output": pass_1_output.model_dump(mode="json"),
        })

        yield _event(job_id, stage="completed", status="completed",
                     partial_result={"phase": "architect_complete"})

    except Exception as exc:  # noqa: BLE001 — surfacing stable error code in SSE
        yield _event(job_id, stage="failed", status="error",
                     error={"code": _stable_error_code(exc), "message": str(exc)})
        raise  # re-raise so the FastAPI handler logs the stack trace via observability.py
```

`_stable_error_code` maps known exception types (`StageAError`, `Pass0Error`, `Pass1Error`, `CoverageError`) to stable string codes (`STAGE_A_FAILED`, `PASS_0_HARD_FAIL`, etc.) so the CLI / frontend can render localized error messages without parsing free-text.

```python
from __future__ import annotations
from typing import AsyncIterator
from fastapi import APIRouter, HTTPException, Request
from fastapi.responses import StreamingResponse
import re

from mcpgen_engine.pipeline import run_pipeline, GenerationSseEvent

router = APIRouter()

# GEN_ID_REGEX from packages/contracts/src/idempotency.ts (Phase 1 D-11)
# Format: gen_<26-char Crockford ULID>
GEN_ID_REGEX = re.compile(r"^gen_[0-9A-HJKMNP-TV-Z]{26}$")


# In-memory job table — Phase 2 single-process. Phase 6+ swaps to Postgres
# `generations` table (already in `infrastructure/neon/migrations/0001_init.sql`
# from Phase 1 D-08).
_JOB_TABLE: dict[str, dict] = {}


@router.post("/api/v1/generate", status_code=202)
async def generate(req: Request) -> dict:
    """Phase 2: accept job, return 202 + sse_url (D-47)."""
    body = await req.json()
    spec_url = body.get("spec_url")
    spec_content = body.get("spec_content")
    options_raw = body.get("options", {})

    job_id = req.headers.get("Idempotency-Key", "")
    if not GEN_ID_REGEX.match(job_id):
        raise HTTPException(status_code=400, detail="invalid Idempotency-Key — expected gen_<ULID>")

    if not spec_url and not spec_content:
        raise HTTPException(status_code=400, detail="must provide spec_url or spec_content")

    _JOB_TABLE[job_id] = {
        "spec_url": spec_url,
        "spec_content": spec_content,
        "options": options_raw,
    }
    return {"job_id": job_id, "sse_url": f"/api/v1/generate/{job_id}/stream"}


@router.get("/api/v1/generate/{job_id}/stream")
async def stream(job_id: str, request: Request) -> StreamingResponse:
    """SSE stream — hand-rolled, mirrors Phase-1 frozen contract."""
    job = _JOB_TABLE.get(job_id)
    if job is None:
        raise HTTPException(status_code=404, detail=f"unknown job {job_id}")

    last_event_id = request.headers.get("Last-Event-ID", "")

    return StreamingResponse(
        _sse_generator(job_id, job, last_event_id),
        media_type="text/event-stream",
        headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no"},
    )


async def _sse_generator(job_id: str, job: dict, last_event_id: str) -> AsyncIterator[bytes]:
    """Format: id: <ULID>\\nevent: <stage>\\ndata: <json>\\n\\n (Phase-1 D-09)."""
    options = _build_generation_options(job["options"])
    async for event in run_pipeline(
        spec_url=job["spec_url"],
        spec_content=job["spec_content"],
        options=options,
        job_id=job_id,
    ):
        if last_event_id and event.event_id <= last_event_id:
            continue  # already delivered — Last-Event-ID resume
        payload = (
            f"id: {event.event_id}\n"
            f"event: {event.stage}\n"
            f"data: {event.model_dump_json()}\n\n"
        )
        yield payload.encode("utf-8")
```
