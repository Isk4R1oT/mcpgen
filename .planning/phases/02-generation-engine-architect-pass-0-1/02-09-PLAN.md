```typescript
import { spawn, type Subprocess } from 'bun';
import { existsSync } from 'node:fs';
import { resolve } from 'node:path';
import { execSync } from 'node:child_process';

const HEALTH_URL = 'http://localhost:8000/health';
const SPAWN_CMD: string[] = [
  'uv', 'run', '--directory', 'apps/generation-engine',
  'uvicorn', 'mcpgen_engine.main:app', '--port', '8000',
];

let _activeProc: Subprocess | null = null;

export async function ensureEngineRunning(): Promise<Subprocess | null> {
  // 1. Check if already running
  try {
    const resp = await fetch(HEALTH_URL, { signal: AbortSignal.timeout(2000) });
    if (resp.ok) return null;
  } catch {
    // not running — try spawn
  }

  // 2. Detect monorepo root (T-2-09-01 mitigation)
  const monorepoRoot = await detectMonorepoRoot();
  if (!monorepoRoot) {
    console.error('Engine not running at http://localhost:8000.');
    console.error('Run `pnpm dev:engine` first, or run `mcpgen init` from the monorepo.');
    process.exit(1);
  }

  // 3. Spawn engine
  const proc = spawn(SPAWN_CMD, {
    cwd: monorepoRoot,
    stdout: 'pipe',
    stderr: 'pipe',
    env: process.env,
  });
  _activeProc = proc;

  // 4. Health-poll (5s max — T-2-09-04 mitigation)
  for (let attempt = 0; attempt < 50; attempt++) {
    await new Promise((r) => setTimeout(r, 100));
    try {
      const resp = await fetch(HEALTH_URL, { signal: AbortSignal.timeout(500) });
      if (resp.ok) return proc;
    } catch {
      // not ready yet
    }
  }

  proc.kill();
  _activeProc = null;
  throw new Error('Engine failed to start within 5s. Check logs from `pnpm dev:engine` directly.');
}


async function detectMonorepoRoot(): Promise<string | null> {
  try {
    const root = execSync('git rev-parse --show-toplevel', { encoding: 'utf-8' }).trim();
    // Defense-in-depth: verify the engine package exists
    if (existsSync(resolve(root, 'apps/generation-engine/pyproject.toml'))) {
      return root;
    }
    return null;
  } catch {
    return null;
  }
}


// Graceful-shutdown: 3 independent cleanup paths (T-2-09-05)
export function registerCleanup(): void {
  const cleanup = () => {
    if (_activeProc) {
      _activeProc.kill('SIGTERM');
      _activeProc = null;
    }
  };
  process.on('SIGINT', () => { cleanup(); process.exit(130); });
  process.on('SIGTERM', () => { cleanup(); process.exit(143); });
  process.on('exit', cleanup);
}
```

**B. `apps/cli/src/init/sse_consumer.ts`** (PATTERNS lines 977-1005):

```typescript
import { createParser, type EventSourceMessage } from 'eventsource-parser';
import type { GenerationSseEvent } from '@mcpgen/contracts';

export async function* consumeSse(url: string): AsyncIterable<GenerationSseEvent> {
  const resp = await fetch(url, { headers: { Accept: 'text/event-stream' } });
  if (!resp.ok) {
    throw new Error(`SSE stream returned ${resp.status} ${resp.statusText}`);
  }
  if (!resp.body) throw new Error('No SSE body');

  const reader = resp.body.getReader();
  const decoder = new TextDecoder();
  const queue: GenerationSseEvent[] = [];
  const parser = createParser({
    onEvent: (msg: EventSourceMessage) => {
      try {
        queue.push(JSON.parse(msg.data) as GenerationSseEvent);
      } catch (err) {
        console.error('Malformed SSE event:', msg.data);
      }
    },
  });

  while (true) {
    const { value, done } = await reader.read();
    if (done) {
      // Drain remaining queue
      while (queue.length) yield queue.shift()!;
      return;
    }
    parser.feed(decoder.decode(value, { stream: true }));
    while (queue.length) yield queue.shift()!;
  }
}
```

  const resolved = resolve(rawDir);
  const cwd = process.cwd();
  // Allow paths under cwd OR absolute paths the user explicitly typed
  // (resolve() already absolutizes; the assertion is symbolic — Phase 2 trusts the user's literal)
  if (resolved !== cwd && !resolved.startsWith(cwd + sep)
      && !rawDir.startsWith('/') && !rawDir.startsWith('~')) {
    throw new Error(`output dir must be under cwd or absolute: ${rawDir}`);

  await mkdir(resolved, { recursive: true, mode: 0o755 });
  return resolved;
}

function renderProgress(event: GenerationSseEvent, s: ReturnType<typeof spinner>): void {
