**A. `passes/pass_1/__init__.py`** — single async orchestrator per D-50:

```python
from __future__ import annotations
import asyncio
from typing import Iterable

from mcpgen_ir.types import Pass0Output, Pass1Output, RawIR, Routing1, SmartId, Tool1, Workflow1
from mcpgen_engine.types import GenerationOptions  # if not yet present, define stub here mirroring Pass 0 UserOptions

from .classify import ClassifiedTools, classify_tool_plans
from .schema_synth import synthesize_universal_tools, synthesize_extra_tool
from .routing import build_routing_config, derive_spec_slug
from .coverage import build_coverage_proof, coverage_pct

PASS_1_SCHEMA_SYNTH_CONCURRENCY = 10  # D-28
MAX_COVERAGE_RETRIES = 3              # D-34


class Pass1Error(ValueError):
    """User-facing Pass 1 failure with stable error code."""


async def run(
    pass_0_output: Pass0Output,
    raw_ir: RawIR,
    options: GenerationOptions,
) -> Pass1Output:
    # Phase 1.1: deterministic classification
    classified = classify_tool_plans(
        pass_0_output.tool_plans,
        composite_candidates=pass_0_output.composite_candidates,
        dependency_graph=raw_ir.dependency_graph,
    )

    # Phase 1.2: schema synthesis — concurrency 10
    sem = asyncio.Semaphore(PASS_1_SCHEMA_SYNTH_CONCURRENCY)

    async def synth_one_universal() -> list[Tool1]:
        async with sem:
            return await synthesize_universal_tools(classified.universal, raw_ir)

    async def synth_one_extra(extra) -> Tool1:
        async with sem:
            return await synthesize_extra_tool(extra, raw_ir)

    universal_tools, *extras = await asyncio.gather(
        synth_one_universal(),
        *[synth_one_extra(e) for e in classified.extras],
    )

    # Phase 1.3: routing (deterministic — no LLM)
    spec_slug = derive_spec_slug(raw_ir.spec_title)
    routing = build_routing_config(universal_tools, extras, spec_slug, raw_ir)

    # Phase 1.4: coverage validation + 3-retry orchestration (D-34)
    proofs = build_coverage_proof(pass_0_output.tool_plans, routing, raw_ir)
    coverage = coverage_pct(pass_0_output.tool_plans, proofs)

    retries = 0
    while coverage < 100.0 and retries < MAX_COVERAGE_RETRIES:
        retries += 1
        uncovered = _find_uncovered(pass_0_output.tool_plans, proofs)
        # re-synth the universal tools with uncovered endpoints in prompt
        universal_tools = await synthesize_universal_tools(
            classified.universal, raw_ir, uncovered=uncovered,
        )
        routing = build_routing_config(universal_tools, extras, spec_slug, raw_ir)
        proofs = build_coverage_proof(pass_0_output.tool_plans, routing, raw_ir)
        coverage = coverage_pct(pass_0_output.tool_plans, proofs)

    if coverage < 100.0:
        # Degraded fallback per D-34 — emit uncovered as specialized_tools with degraded warning
        extras = list(extras) + _degrade_uncovered_to_specialized(
            uncovered=_find_uncovered(pass_0_output.tool_plans, proofs),
            raw_ir=raw_ir,
        )
        routing = build_routing_config(universal_tools, extras, spec_slug, raw_ir)
        proofs = build_coverage_proof(pass_0_output.tool_plans, routing, raw_ir)
        coverage = coverage_pct(pass_0_output.tool_plans, proofs)

    return Pass1Output(
        tools=[*universal_tools, *extras],
        routing=routing,
        workflows=[],  # Phase 2 emits empty; Phase 4 Stage E generates handler bodies
        coverage_pct=coverage,
    )
```

def build_smart_id_format(spec_slug: str) -> str:
    """D-31 — schema-level format. Phase 6 prepends `{tenant_short_id}-` literal."""
    # Returns the format with `{spec_slug}` substituted to its concrete value
    # but `{type}`, `{collection}`, `{identifier}` left as Python format placeholders
    # so Stage E codegen can use Python str.format() at runtime.

def build_smart_id_regex(spec_slug: str, types: Sequence[str], collections: Sequence[str]) -> str:
    """D-56 — for fixture round-trip test (Pitfall #1).

    schema-level regex matches IDs from any deployment. Two synthetic tenants
    `acme-` + `widgets-` produce literally different IDs that both match.

    return rf"^[a-z0-9-]+-{re.escape(spec_slug)}:({type_alt}):({coll_alt}):[A-Za-z0-9_./-]+$"
