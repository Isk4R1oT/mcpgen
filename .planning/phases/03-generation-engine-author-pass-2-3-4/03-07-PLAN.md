      ```python
      class FilterStrategy(str, Enum):
          STRUCTURED_OBJECT = "A"
          DSL_STRING = "B"
          INDIVIDUAL_PARAMS = "C"

      # D-18 / Pass 3 design §11.1 — frozen operator enum (do NOT extend without docs/decisions/ entry)
      STRUCTURED_OBJECT_OPERATORS: Final[tuple[str, ...]] = (
          "Equal", "NotEqual", "GreaterThan", "LessThan",
          "GreaterThanOrEqual", "LessThanOrEqual",
          "Contains", "In", "NotIn",
      )

      _INDIVIDUAL_PARAMS_THRESHOLD: Final[int] = 4  # D-18 — "≤4 filter params"

      _NATIVE_QUERY_HINT_REGEX: Final[re.Pattern[str]] = re.compile(
          r"\b(JQL|GraphQL where clause|SQL filter|GraphQL filter|GraphQL query)\b",
          re.IGNORECASE,
      )

      _SIMPLE_TYPES: Final[frozenset[str]] = frozenset({"string", "integer", "boolean", "number"})
      ```
    - Implement private helpers:
      ```python
      def _has_native_query_language(raw_ir: RawIR) -> bool:
          """True if any endpoint hints at JQL / GraphQL / SQL filter native query.

          Two signals checked:
          1. Vendor extension `x-query-language` set on any endpoint (defensive lookup —
             Phase 2 D-51 may store extensions in a sidecar Mapping; if attribute is
             absent on Endpoint, this signal is False — that's acceptable per D-18
             which says hint via description text is also sufficient).
          2. Endpoint description matches _NATIVE_QUERY_HINT_REGEX.
          """
          for ep in raw_ir.endpoints:
              # Signal 1: vendor extension (defensive lookup)
              extensions = getattr(ep, "extensions", None) or {}
              if isinstance(extensions, dict) and "x-query-language" in extensions:
                  return True
              # Signal 2: description text
              if ep.description and _NATIVE_QUERY_HINT_REGEX.search(ep.description):
                  return True
          return False

      def _is_simple_filter_param(param: ParameterSpec) -> bool:
          """True if param looks like simple equality (scalar type, no pattern, no nested struct)."""
          return (
              param.type in _SIMPLE_TYPES
              and param.pattern is None
              # Object/array types disqualify; covered by the type check above.
          )

      def _collect_filter_params(extracted: dict[str, list[ParameterSpec]]) -> list[ParameterSpec]:
          """Flatten all filter-flagged params across all tools (D-18 server-scope analysis)."""
          out: list[ParameterSpec] = []
          for tool_name, params in extracted.items():
              for p in params:
                  if p.is_filter:
                      out.append(p)
          return out
      ```
    - Implement public selector:
      ```python
      def detect_filter_strategy(
          raw_ir: RawIR, extracted: dict[str, list[ParameterSpec]],
      ) -> FilterStrategy:
          """Decide one filter strategy per server per D-18 verbatim decision tree."""
          # 1. Native query language hint → DSL_STRING
          if _has_native_query_language(raw_ir):
              chosen = FilterStrategy.DSL_STRING
              _log.info(
                  "pass_3.filter_design.detect", strategy=chosen.value,
                  reason="native_query_language_hint",
              )
              return chosen

          # 2. Few simple filter params → INDIVIDUAL_PARAMS
          filter_params = _collect_filter_params(extracted)
          if (
              len(filter_params) <= _INDIVIDUAL_PARAMS_THRESHOLD
              and all(_is_simple_filter_param(p) for p in filter_params)
              and len(filter_params) > 0  # if zero filter params we still default to STRUCTURED_OBJECT
          ):
              chosen = FilterStrategy.INDIVIDUAL_PARAMS
              _log.info(
                  "pass_3.filter_design.detect", strategy=chosen.value,
                  reason="simple_few_filter_params", filter_param_count=len(filter_params),
              )
              return chosen

          # 3. DEFAULT — structured object
          chosen = FilterStrategy.STRUCTURED_OBJECT
          _log.info(
              "pass_3.filter_design.detect", strategy=chosen.value,
              reason="default", filter_param_count=len(filter_params),
          )
          return chosen
      ```
    - Implement public emitter:
      ```python
      def emit_filter_schema(
          strategy: FilterStrategy,
          extracted_filter_params: list[ParameterSpec],
      ) -> dict[str, Any]:
          """Emit the JSON Schema fragment for the `filter` parameter under the chosen strategy.

          Returns:
            A: dict with {type:'object', properties:{property,operator,value}, additionalProperties:false}
            B: dict with {type:'string', description:'...DSL...'}
            C: dict with {type:'object', properties:{<one per simple filter param>}, additionalProperties:false}
              — caller is expected to lift these as TOP-LEVEL inputSchema properties (NOT nest under 'filter')
          """
          if strategy == FilterStrategy.STRUCTURED_OBJECT:
              return {
                  "type": "object",
                  "description": (
                      "Structured filter object with property + operator + value. "
                      f"Supported operators: {', '.join(STRUCTURED_OBJECT_OPERATORS)}."
                  ),
                  "properties": {
                      "property": {
                          "type": "string",
                          "description": "Field name to filter on.",
                      },
                      "operator": {
                          "type": "string",
                          "enum": list(STRUCTURED_OBJECT_OPERATORS),
                          "description": "Comparison operator.",
                      },
                      "value": {
                          "description": "Comparison value (type depends on the field).",
                      },
                  },
                  "required": ["property", "operator", "value"],
                  "additionalProperties": False,
              }

          if strategy == FilterStrategy.DSL_STRING:
              return {
                  "type": "string",
                  "description": (
                      "Native query string in the upstream API's query language "
                      "(e.g., JQL / GraphQL `where` / SQL filter). "
                      "Refer to the upstream API documentation for syntax."
                  ),
              }

          # FilterStrategy.INDIVIDUAL_PARAMS
          # Caller lifts these as top-level inputSchema properties; this returns the
          # "envelope" indicating the caller should NOT nest under 'filter'.
          properties: dict[str, dict[str, Any]] = {}
          for p in extracted_filter_params:
              if not _is_simple_filter_param(p):
                  continue
              entry: dict[str, Any] = {"type": p.type}
              if p.enum is not None:
                  entry["enum"] = p.enum
              if p.description:
                  entry["description"] = p.description[:300]  # bound to keep schema lean
              if p.default is not None:
                  entry["default"] = p.default
              properties[p.name] = entry
          return {
              "type": "object",
              "description": "Individual filter parameters (lift to top-level inputSchema per D-18 strategy C).",
              "properties": properties,
              "additionalProperties": False,
              "_individual_params_marker": True,  # sentinel for Plan 03-09 caller to lift to top-level
          }
      ```
