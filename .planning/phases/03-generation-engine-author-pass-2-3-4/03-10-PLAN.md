      ```python
      def apply_tool_type_rules(tool: Tool1) -> RuleResult:
          """Look up tool in _TOOL_TYPE_RULES; return RuleResult.

          Returns is_decisive=False for action / workflow tools (not in table —
          caller routes to verb pattern OR conservative aggregation).
          """
          if tool.type == Type.universal:
              if tool.universal_tool is None:
                  # Universal but no UniversalTool tag — defensive (shouldn't happen)
                  return RuleResult(is_decisive=False)
              key = (tool.type.value, tool.universal_tool.value)
          elif tool.type == Type.specialized:
              key = (tool.type.value, None)
          else:
              # action / workflow — not in this table
              return RuleResult(is_decisive=False)

          if key not in _TOOL_TYPE_RULES:
              return RuleResult(is_decisive=False)

          read_only, destructive, idempotent = _TOOL_TYPE_RULES[key]
          return RuleResult(
              is_decisive=True,
              read_only=read_only,
              destructive=destructive,
              idempotent=idempotent,
          )
      ```

      ```python
      def aggregate_workflow_annotations(
          workflow: Tool1, all_tools: list[Tool1],
      ) -> RuleResult:
          """D-30: workflow conservative aggregation across sub-operations.

          - readOnly = AND across sub-tool annotations (any non-readOnly → workflow non-readOnly)
          - destructive = OR across subs (any destructive → workflow destructive)
          - idempotent = AND across subs (any non-idempotent → workflow non-idempotent)

          Sub-operations: for each endpoint in workflow.source_endpoints, find the
          OTHER tool that ALSO has that endpoint in its source_endpoints (the
          "owning" tool for that endpoint). Apply that tool's rules recursively.

          If a sub-endpoint isn't claimed by any other tool, fall back to
          CONSERVATIVE per D-29: (False, True, False) — i.e., not readOnly,
          destructive, not idempotent — UX safety > optimization.
          """
          if workflow.type != Type.workflow:
              # defensive — caller should have filtered to workflows only
              return RuleResult(is_decisive=False)

          read_onlys: list[bool] = []
          destructives: list[bool] = []
          idempotents: list[bool] = []

          conservative_default = (False, True, False)  # D-29 fallback (UX safety)

          # Build endpoint→owning-tool index (excluding the workflow itself)
          endpoint_owner: dict[str, Tool1] = {}
          for t in all_tools:
              if t.name == workflow.name:
                  continue
              for ep in t.source_endpoints:
                  endpoint_owner.setdefault(ep, t)  # first claimer wins

          for endpoint_id in workflow.source_endpoints:
              owner = endpoint_owner.get(endpoint_id)
              if owner is None:
                  # Conservative fallback
                  ro, dest, idem = conservative_default
              elif owner.type == Type.workflow:
                  # Recursive — but bounded: skip self-references / cycles by treating
                  # nested workflow as conservative default (avoid infinite recursion)
                  ro, dest, idem = conservative_default
              else:
                  # Apply owner's rule (rules table for universal/specialized; verb pattern
                  # for action — but verb pattern is in verbs.py module; for aggregation
                  # purposes we use a LOCAL conservative default for action since we don't
                  # want a circular import. Plan 03-11 caller can override by re-running
                  # aggregate after action verbs resolved.)
                  if owner.type in (Type.universal, Type.specialized):
                      sub_result = apply_tool_type_rules(owner)
                      if sub_result.is_decisive:
                          ro = sub_result.read_only or False
                          dest = sub_result.destructive or False
                          idem = sub_result.idempotent or False
                      else:
                          ro, dest, idem = conservative_default
                  else:
                      # action — conservative for aggregation (Plan 03-11 will refine if needed)
                      ro, dest, idem = conservative_default
              read_onlys.append(ro)
              destructives.append(dest)
              idempotents.append(idem)

          if not read_onlys:
              # Empty workflow — conservative
              return RuleResult(
                  is_decisive=True, read_only=False, destructive=True, idempotent=False,
              )

          return RuleResult(
              is_decisive=True,
              read_only=all(read_onlys),       # AND
              destructive=any(destructives),   # OR
              idempotent=all(idempotents),     # AND
          )
      ```

      ```python
      def generate_title(tool: Tool1) -> str:
          """Generate human-readable title per D-31."""
          parts = tool.name.split("_")

          if tool.type == Type.action and len(parts) >= 2:
              # Verb reordering: last token = verb, others = object
              # Example: "charges_capture" → "Capture Charge"
              verb = parts[-1]
              obj_parts = parts[:-1]
              obj = " ".join(p.capitalize() for p in obj_parts)
              # Singularize trailing 's' on the object's last segment for grammatical fit:
              # "charges_capture" → "Capture Charge" (singular). Per D-31, simple title-case;
              # de-pluralization is a polish nicety — apply when object ends in 's' AND
              # has > 3 chars (avoid stripping "tags" → "tag" which is fine but "ws" → "w" isn't).
              if obj_parts and obj_parts[-1].endswith("s") and len(obj_parts[-1]) > 3:
                  # Replace trailing s on the last word in `obj`
                  last_word = obj_parts[-1].capitalize()
                  if obj.endswith(last_word):
                      obj = obj[: -len(last_word)] + last_word[:-1]
              title = f"{verb.capitalize()} {obj}".strip()
          elif tool.type == Type.workflow:
              # Title-cased with verb naturally at start in workflow names
              title = " ".join(p.capitalize() for p in parts)
          else:
              # Universal / specialized — straight title-case
              title = " ".join(p.capitalize() for p in parts)

          if len(title) > _MAX_TITLE_LENGTH:
              # Truncate with ellipsis (use single … char for terminal compatibility)
              title = title[: _MAX_TITLE_LENGTH - 1].rstrip() + "…"
          return title
      ```
