      def is_within_budget(text: str, tool_type: str) -> tuple[bool, str | None]:
          """Returns (within_budget, retry_hint) — hint is None when OK."""
          if tool_type not in LENGTH_BUDGETS:

          if n > max_t:

          return True, None
