      ```python
      """Pass 4 — system prompt + user-prompt builder for selective LLM judgment.

      Used ONLY for medium-confidence verb tools (typically 0-3 per server per
      Plan 03-10's `match_verb_pattern` returning confidence='medium').

      Per Research A3, the LLM judge is asked for ONLY the 3 mutable booleans
      (readOnlyHint, destructiveHint, idempotentHint) + a brief rationale.
      openWorldHint is NEVER requested from the LLM (D-27 invariant — set
      Python-side at IR assembly).

      References:
      - 03-CONTEXT.md D-26 (Phase 2) + D-27 + D-29
      - 03-RESEARCH.md §"Phase-3-specific subtleties" A3
      - docs/mcpgen-pass-4-design.md §6
      """
      from __future__ import annotations
      from typing import Final

      PASS_4_JUDGE_SYSTEM_PROMPT: Final[str] = """You are a quality judge for MCP tool annotations (Pass 4).

      For the given tool, determine 3 boolean hints describing its semantics:
      - readOnlyHint: true if the tool ONLY READS data (no upstream state changes); false otherwise.
      - destructiveHint: true if the tool may permanently delete/destroy data; false otherwise.
      - idempotentHint: true if calling the tool twice with same args produces the same result as calling once; false otherwise.

      RULES:
      - Be conservative: when in doubt, prefer `readOnly=false, destructive=true, idempotent=false` (UX safety).
      - readOnly=true and destructive=true is a contradiction; if you would set both, set readOnly=false.
      - readOnly=true implies idempotent=true (reads are inherently idempotent); but the consistency phase auto-fixes this — you can leave them inconsistent and the validator will reconcile.
      - Do NOT include openWorldHint in your response. The framework adds it.

      Provide a brief rationale (1-2 sentences) explaining your reasoning.
      """

      def build_judge_prompt(tool_name: str, tool_description: str | None) -> str:
          """Build user prompt — short, deterministic, no untrusted spec content."""
          desc_block = ""
          if tool_description:
              # Bound to keep prompts cheap; description came from Pass 2 (already validated)
              desc_block = f"\n\nDescription excerpt:\n{tool_description[:500]}"
          return (
              f"Tool name: {tool_name}\n"
              f"Tool category: action (medium-confidence verb pattern)"
              f"{desc_block}\n\n"
              f"Determine readOnlyHint, destructiveHint, idempotentHint per the system prompt."
          )
      ```
    - **llm_judge.py:**
      ```python
      """Pass 4 — selective LLM judgment for medium-confidence action verbs (D-26 Phase 2).

      Concurrency Sem(5) per D-26. Single retry on Pydantic ValidationError.
      Falls back to D-29 conservative defaults after retry exhaustion (NOT a
      Pass4Error raise — emit-and-continue).

      `_LlmJudgeOutput` deliberately OMITS openWorldHint per Research A3 +
      D-27 invariant; openWorldHint constructed Python-side at IR assembly
      in consistency.assemble_annotations_with_open_world_hint.

      References:
      - 03-CONTEXT.md D-26 (Phase 2) + D-27 + D-29
      - 03-RESEARCH.md §"Phase-3-specific subtleties" A3
      - 03-PATTERNS.md `passes/pass_4/llm_judge.py` row
      """
      from __future__ import annotations
      import asyncio
      from typing import Final
      import httpx
      import structlog
      from pydantic import BaseModel, ConfigDict, Field, ValidationError
      from pydantic_ai import Agent
      from pydantic_ai.exceptions import UnexpectedModelBehavior
      from mcpgen_ir.types import Pass2Output
      from mcpgen_engine.llm.agent_factory import make_agent
      from mcpgen_engine.llm.sampling import PASS_4_SETTINGS
      from mcpgen_engine.passes.pass_4.prompts import (
          PASS_4_JUDGE_SYSTEM_PROMPT, build_judge_prompt,
      )

      _log = structlog.get_logger(__name__)

      PASS_4_JUDGE_CONCURRENCY: Final[int] = 5  # D-26 Phase 2
      _MAX_TRANSIENT_RETRIES: Final[int] = 3
      _MAX_VALIDATION_RETRIES: Final[int] = 1  # D-29: 1 retry then conservative fallback
      _TRANSIENT_BACKOFF_BASE: Final[float] = 1.0
      _TRANSIENT_BACKOFF_MAX: Final[float] = 4.0

      # D-29 conservative defaults — UX safety > optimization
      _CONSERVATIVE_DEFAULTS: Final[dict[str, bool]] = {
          "readOnlyHint": False,
          "destructiveHint": True,
          "idempotentHint": False,
      }


      class _LlmJudgeOutput(BaseModel):
          """LLM-emitted output — 3 booleans + rationale.

          openWorldHint DELIBERATELY OMITTED per Research A3 + D-27. The IR
          construction site (consistency.assemble_annotations_with_open_world_hint)
          adds openWorldHint=True Python-side.
          """
          model_config = ConfigDict(extra="forbid")
          readOnlyHint: bool
          destructiveHint: bool
          idempotentHint: bool
          rationale: str = Field(default="")


      PASS_4_JUDGE_AGENT: Final[Agent[None, _LlmJudgeOutput]] = make_agent(
          output_type=_LlmJudgeOutput, system_prompt=PASS_4_JUDGE_SYSTEM_PROMPT,
      )


      async def _run_with_transient_retry(prompt: str) -> _LlmJudgeOutput:
          last_exc: BaseException | None = None
          for attempt in range(_MAX_TRANSIENT_RETRIES):
              try:
                  result = await PASS_4_JUDGE_AGENT.run(
                      prompt, model_settings=PASS_4_SETTINGS,
                  )
                  return result.output
              except httpx.HTTPError as exc:
                  last_exc = exc
                  delay = min(_TRANSIENT_BACKOFF_BASE * (2 ** attempt), _TRANSIENT_BACKOFF_MAX)
                  _log.warning(
                      "pass_4.judge.transient_retry",
                      attempt=attempt + 1, max_attempts=_MAX_TRANSIENT_RETRIES,
                      delay_s=delay, error_class=type(exc).__name__,
                  )
                  await asyncio.sleep(delay)
          assert last_exc is not None
          raise last_exc


      async def _judge_one(tool_name: str, tool_description: str | None) -> dict[str, bool]:
          """Judge a single medium-confidence action tool. Returns the 3 booleans
          (or _CONSERVATIVE_DEFAULTS on failure)."""
          for attempt in range(_MAX_VALIDATION_RETRIES + 1):  # 1 + 1 retry
              prompt = build_judge_prompt(tool_name, tool_description)
              try:
                  output = await _run_with_transient_retry(prompt)
                  return {
                      "readOnlyHint": output.readOnlyHint,
                      "destructiveHint": output.destructiveHint,
                      "idempotentHint": output.idempotentHint,
                  }
              except (ValidationError, UnexpectedModelBehavior, httpx.HTTPError) as exc:
                  _log.warning(
                      "pass_4.judge.validation_or_transient_failed",
                      tool_name=tool_name, attempt=attempt + 1,
                      error_class=type(exc).__name__,
                  )
          _log.warning(
              "pass_4.judge.fallback_to_conservative_defaults",
              tool_name=tool_name,
          )
          return dict(_CONSERVATIVE_DEFAULTS)


      async def judge_action_tools(
          needs_llm_review: list[str],
          pass_2_output: Pass2Output | None,
      ) -> dict[str, dict[str, bool]]:
          """Selective LLM judgment for medium-confidence action tools.

          Concurrency Sem(5). Returns {tool_name: {readOnlyHint, destructiveHint, idempotentHint}}.
          """
          if not needs_llm_review:
              return {}
          sem = asyncio.Semaphore(PASS_4_JUDGE_CONCURRENCY)

          # Look up tool descriptions from Pass 2 output for richer prompt context
          descriptions_map: dict[str, str | None] = {}
          if pass_2_output is not None:
              for tool_name in needs_llm_review:
                  desc = pass_2_output.descriptions.get(tool_name)
                  if desc is not None:
                      # Pass2Output.descriptions value is `Descriptions` (with description_hash);
                      # render to a short prompt-friendly string from purpose only.
                      descriptions_map[tool_name] = desc.purpose
                  else:
                      descriptions_map[tool_name] = None
          else:
              descriptions_map = {n: None for n in needs_llm_review}

          async def _bound(name: str) -> tuple[str, dict[str, bool]]:
              async with sem:
                  triple = await _judge_one(name, descriptions_map.get(name))
                  return name, triple

          pairs = await asyncio.gather(*(_bound(n) for n in needs_llm_review))
          return dict(pairs)
      ```

      ```python
      async def run(
          pass_3_output: Pass3Output,
          pass_2_output: Pass2Output,
          pass_1_output: Pass1Output,
      ) -> Pass4Output:
          """3-phase orchestrator per D-26."""
          start = time.monotonic()
          all_tools = pass_1_output.tools

          # ===== Phase 1: deterministic rules + verb matching + title generation =====
          triples: dict[str, dict[str, bool]] = {}  # {tool_name: {readOnlyHint, destructiveHint, idempotentHint}}
          titles: dict[str, str] = {}
          needs_llm_review: list[str] = []

          for tool in all_tools:
              titles[tool.name] = generate_title(tool)

              # Try tool-type rules first (covers universal + specialized)
              rule_result = apply_tool_type_rules(tool)
              if rule_result.is_decisive:
                  triples[tool.name] = {
                      "readOnlyHint": rule_result.read_only or False,
                      "destructiveHint": rule_result.destructive or False,
                      "idempotentHint": rule_result.idempotent or False,
                  }
                  continue

              # Action: try verb pattern
              if tool.type == Type.action:
                  verb_result = match_verb_pattern(tool.name)
                  if verb_result.confidence == "high":
                      triples[tool.name] = {
                          "readOnlyHint": verb_result.read_only or False,
                          "destructiveHint": verb_result.destructive or False,
                          "idempotentHint": verb_result.idempotent or False,
                      }
                      continue
                  # medium or none → LLM review (or conservative fallback if LLM disabled)
                  needs_llm_review.append(tool.name)
                  continue

              # Workflow: conservative aggregation
              if tool.type == Type.workflow:
                  agg_result = aggregate_workflow_annotations(tool, all_tools)
                  triples[tool.name] = {
                      "readOnlyHint": agg_result.read_only or False,
                      "destructiveHint": agg_result.destructive or False,
                      "idempotentHint": agg_result.idempotent or False,
                  }
                  continue

              # Defensive: any other type → conservative defaults (D-29)
              triples[tool.name] = dict(_CONSERVATIVE_DEFAULTS)

          # ===== Phase 2: selective LLM judgment for medium-confidence actions =====
          if needs_llm_review:
              llm_results = await judge_action_tools(needs_llm_review, pass_2_output)
              for name, result_triple in llm_results.items():
                  triples[name] = result_triple

          # Any tools still missing (defensive — should be 0 here) → conservative defaults
          for tool in all_tools:
              if tool.name not in triples:
                  _log.warning(
                      "pass_4.run.fallback_to_conservative_for_missing",
                      tool_name=tool.name,
                  )
                  triples[tool.name] = dict(_CONSERVATIVE_DEFAULTS)

          # ===== Phase 3: consistency validation with auto-fix + IR assembly =====
          consistent = enforce_consistency_with_autofix(triples)
          annotations = assemble_annotations_with_open_world_hint(consistent)

          elapsed_ms = int((time.monotonic() - start) * 1000)
          _log.info(
              "pass_4.run.complete",
              tool_count=len(annotations),
              llm_review_count=len(needs_llm_review),
              elapsed_ms=elapsed_ms,
          )

          return Pass4Output(annotations=annotations, titles=titles)
      ```
