      def validate_input_schema(tool_name: str, schema: dict[str, Any]) -> dict[str, Any]:
          """D-22: enforce additionalProperties: false; validate schema validity.

          Returns the (possibly auto-injected) schema. Raises Pass3Error on hard
          violations: explicit additionalProperties=true, OR Draft 2020-12 SchemaError.
          """
          # D-22 hard error: explicit true is intentional disagreement with policy
          if schema.get("additionalProperties") is True:
              raise Pass3Error(
                  f"INVALID_SCHEMA: tool '{tool_name}' has additionalProperties=true; "
                  f"must be false per D-22",
                  violations=["additionalProperties=true"],
              )
          # D-22 auto-inject: omitted is treated as policy-conformance candidate
          schema = dict(schema)  # avoid mutating caller
          schema.setdefault("additionalProperties", False)
          # Validate the schema itself is well-formed
          try:
              jsonschema.Draft202012Validator.check_schema(schema)
          except SchemaError as exc:
              raise Pass3Error(
                  f"INVALID_SCHEMA: tool '{tool_name}': {exc.message}",
                  violations=[f"schema_error: {exc.message}"],
              ) from exc
          return schema

      def validate_smart_id_pattern(
          tool_name: str, schema: dict[str, Any], expected_pattern: str,
      ) -> None:
          """D-20: every property where the source ParameterSpec had is_smart_id=True
          MUST have the expected smart-ID pattern in the JSON Schema.

          Identifies smart-ID properties by name (matches `^id$` or `^[a-z][a-z0-9_]*_id$`)
          since the JSON Schema doesn't carry the original is_smart_id flag.
          """
          import re
          smart_id_name_re = re.compile(r"^(id|[a-z][a-z0-9_]*_id)$", re.IGNORECASE)
          properties = schema.get("properties", {}) or {}
          mismatches: list[str] = []
          for prop_name, prop_schema in properties.items():
              if smart_id_name_re.match(prop_name) and isinstance(prop_schema, dict):
                  actual_pattern = prop_schema.get("pattern")
                  if actual_pattern != expected_pattern:
                      mismatches.append(
                          f"{prop_name}: expected pattern {expected_pattern!r} got {actual_pattern!r}"
                      )
          if mismatches:
              raise Pass3Error(
                  f"SMART_ID_DRIFT: tool '{tool_name}': {'; '.join(mismatches)}",
                  violations=mismatches,
              )

    - Implement `validate_filter_consistency(input_schemas, chosen_strategy)`:
      ```python
      def validate_filter_consistency(
          input_schemas: dict[str, dict[str, Any]], chosen_strategy: FilterStrategy,
      ) -> None:
          """D-18: every tool with a `filter` param must emit a shape consistent
          with the chosen strategy across the entire server.

          Strategy A → filter is `{type: 'object', properties: {property, operator, value}}`
          Strategy B → filter is `{type: 'string'}`
          Strategy C → filter param is NOT present at top-level (individual params lifted)
          """
          drifts: list[str] = []
          for tool_name, schema in input_schemas.items():
              properties = schema.get("properties", {}) or {}
              filter_prop = properties.get("filter")
              if filter_prop is None:
                  continue  # OK — tool doesn't have a filter
              if chosen_strategy == FilterStrategy.STRUCTURED_OBJECT:
                  if (filter_prop.get("type") != "object"
                          or "operator" not in (filter_prop.get("properties") or {})):
                      drifts.append(f"{tool_name}: filter not structured-object shape")
              elif chosen_strategy == FilterStrategy.DSL_STRING:
                  if filter_prop.get("type") != "string":
                      drifts.append(f"{tool_name}: filter not string DSL")
              elif chosen_strategy == FilterStrategy.INDIVIDUAL_PARAMS:
                  # Strategy C lifts to top-level; presence of `filter` indicates drift
                  drifts.append(f"{tool_name}: filter param present under INDIVIDUAL_PARAMS strategy")
          if drifts:
              raise Pass3Error(
                  f"FILTER_INCONSISTENT: chosen={chosen_strategy.value}; {'; '.join(drifts)}",
                  violations=drifts,
              )
      ```

          for tool_name, params in extracted.items():
              # Determine if this tool is a list-filter context (universal list_objects)
              is_list_filter = tool_name == "list_objects"
              normalized[tool_name] = normalize_all_param_names(params, is_list_filter_context=is_list_filter)

          # Phase 2 (LLM ‖×20 pipeline-scoped): enrich params
          tool_types_by_name = {t.name: t.type.value for t in pass_1_output.tools}
          enriched = await enrich_all_params(normalized, tool_types_by_name)

          # Detect filter strategy (D-18 — once per server)
          filter_strategy = detect_filter_strategy(raw_ir, normalized)

          # Slug for smart-ID pattern (D-20)
          spec_slug = slugify_spec_title(spec_title or "spec")
          smart_id_pattern = build_smart_id_pattern_for_param(
              pass_1_output.routing.smart_id, spec_slug,
          )
          smart_id_description = build_smart_id_description(
              pass_1_output.routing.smart_id, spec_slug,
          )

          # Phase 3 assembly + per-tool validation
          tools_by_name = {t.name: t for t in pass_1_output.tools}
          input_schemas: dict[str, dict[str, Any]] = {}
          for tool_name, pairs in enriched.items():
              tool = tools_by_name[tool_name]
              universal_tool_name = (
                  tool.universal_tool.value if tool.universal_tool is not None else None
              )
              # Assemble
              raw_schema = _assemble_input_schema_for_tool(
                  tool_name=tool_name,
                  tool_type=tool.type,
                  universal_tool_name=universal_tool_name,
                  params_with_enrichment=pairs,
                  smart_id_pattern=smart_id_pattern,
                  smart_id_description=smart_id_description,
                  filter_strategy=filter_strategy,
              )
              # Validate (raises Pass3Error on hard violation)
              validate_param_uniqueness(tool_name, [p for p, _ in pairs])
              validate_search_fetch_compliance(tool_name, [p for p, _ in pairs])
              validated_schema = validate_input_schema(tool_name, raw_schema)
              validate_smart_id_pattern(tool_name, validated_schema, smart_id_pattern)
              input_schemas[tool_name] = validated_schema

          # Server-wide filter consistency (D-18)
          validate_filter_consistency(input_schemas, filter_strategy)

          # Phase 4 (LLM ‖×10): inline quality gate
          quality_warnings = await quality_gate_all_tools(input_schemas, pass_1_output)

          warning_count = sum(1 for v in quality_warnings.values() if v)
          elapsed_ms = int((time.monotonic() - start) * 1000)
          _log.info(
              "pass_3.run.complete",
              tool_count=len(input_schemas),
              filter_strategy=filter_strategy.value,
              quality_warning_count=warning_count,
              elapsed_ms=elapsed_ms,
          )

          return Pass3Output(input_schemas=input_schemas)
