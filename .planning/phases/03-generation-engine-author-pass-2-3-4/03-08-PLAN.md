      def normalize_param_name(
          raw: str, entity_hint: str | None, is_list_filter_context: bool = False,
      ) -> str:
          """Apply D-19 rules in order to produce normalized snake_case name.

          Args:
              raw: original param name as it appears in the spec
              entity_hint: entity name from endpoint tags[0] or path segment (used to qualify ambiguous bare names)
              is_list_filter_context: when True, bare 'time' renamed to 'created_at' (D-19); else kept

          Returns:
              normalized snake_case name (lowercase ASCII; never empty)
          """
          name = raw
          # Step 1: camelCase / PascalCase → snake_case
          name = _CAMEL_CASE_REGEX.sub(r"\1_\2", name).lower()
          # Step 2: strip _param / _arg suffix
          name = _PARAM_SUFFIX_REGEX.sub("", name)
          # Step 3: bare 'data' → 'payload'
          if name == "data":
              return _DATA_RENAME
          # Step 4: bare 'time' → 'created_at' (only in list-filter context)
          if name == "time" and is_list_filter_context:
              return _TIME_RENAME
          # Step 5: bare 'id' / 'status' → qualified with entity_hint
          if name in _AMBIGUOUS_BARE_NAMES:
              if entity_hint:
                  return f"{_to_snake(entity_hint)}_{name}"
              return name
          return name

      def _to_snake(text: str) -> str:
          """Lower-case + replace whitespace with underscore (helper for entity_hint normalization)."""
          return re.sub(r"\s+", "_", text.strip()).lower()
      ```

      def normalize_all_param_names(
          params: list[ParameterSpec], is_list_filter_context: bool = False,
      ) -> list[ParameterSpec]:
          """Apply normalize_param_name to every param; resolve collisions per D-19.

          Returns a NEW list (immutable transform — does NOT mutate input).

          Collision rule (D-19): if two params normalize to the same name, the SECOND
          occurrence reverts to its original raw name. If the raw name itself collides
          with an already-seen name, append a digit suffix `_2`, `_3`, ... .
          """
          seen: set[str] = set()
          out: list[ParameterSpec] = []
          for p in params:
              normalized = normalize_param_name(
                  p.name, p.entity_hint, is_list_filter_context=is_list_filter_context,
              )
              if normalized in seen:
                  # Collision: revert to raw name first
                  candidate = p.name
                  if candidate in seen:
                      # Raw also collides — append digit suffix
                      idx = 2
                      while f"{candidate}_{idx}" in seen:
                          idx += 1
                      candidate = f"{candidate}_{idx}"
                  normalized = candidate
              seen.add(normalized)
              out.append(p.model_copy(update={"name": normalized}))
          return out
      ```

      def slugify_spec_title(title: str) -> str:
          """Derive a deterministic spec_slug from RawIR.info.title.

          Lowercase + replace non-alphanumeric (except dash) with dash + collapse
          multi-dash + strip leading/trailing dash + cap at 32 chars.

          Examples:
              "Stripe API"           → "stripe-api"
              "GitHub REST API v3"   → "github-rest-api-v3"
              "Notion API (Beta)"    → "notion-api-beta"
          """
          if not title:
              return "spec"  # defensive default
          slug = title.lower()
          slug = _NON_ALPHANUM_DASH_REGEX.sub("-", slug)
          slug = _MULTIPLE_DASH_REGEX.sub("-", slug)
          slug = slug.strip("-")
