<!-- Wrapper signature per PATTERNS.md lines 793-810 -->
```python
import logfire
from pydantic_ai import Agent
from typing import Any

async def run_with_tracing(
    agent: Agent[Any, Any],
    prompt: str,
    *,
    session_id: str,
    stage: str,
    model_settings: dict[str, Any],
) -> Any:
    """D-06 + Pitfall #2: wrap agent.run with Langfuse-namespaced span attrs."""
    with logfire.span("agent.run", attributes={
        "langfuse.session.id": session_id,
        "langfuse.tags": [stage],
    }):
        return await agent.run(prompt, model_settings=model_settings)
```

<!-- Scrubbing callback per PATTERNS.md lines 750-762 -->
```python
def _preserve_langfuse_session_id(match: logfire.ScrubMatch):
    if match.path == ("attributes", "langfuse.session.id"):
        return match.value
    if match.path == ("attributes", "langfuse.user.id"):
        return match.value
    return None  # let Logfire scrub the rest
```

<!-- Spec content scrubber for >10K chars (D-07) -->
```python
def _scrub_long_spec_attributes(match: logfire.ScrubMatch):
    sensitive_keys = {"spec_yaml", "spec_url_response_body", "raw_ir.openapi", "prompt.system"}
    last_segment = match.path[-1] if match.path else None
    if last_segment in sensitive_keys and isinstance(match.value, str) and len(match.value) > 10_000:
        import hashlib
        sha = hashlib.sha256(match.value.encode("utf-8")).hexdigest()[:16]
        return f"<spec redacted, sha256:{sha}>"
    return None
```
