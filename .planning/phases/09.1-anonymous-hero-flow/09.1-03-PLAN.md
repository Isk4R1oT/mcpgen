```typescript
// apps/api/src/lib/anon-session.ts
import type { Context } from 'hono';
import { getCookie, setCookie } from 'hono/cookie';
import { ulid } from 'ulid';

export const ANON_COOKIE_NAME = 'mcpgen_anon_session';
const ANON_COOKIE_MAX_AGE_SEC = 7 * 24 * 60 * 60; // 7 days per D-04 + D-06
const ULID_REGEX = /^[0-9A-HJKMNP-TV-Z]{26}$/;

export function ensureAnonSession(c: Context, opts: { forceReissue?: boolean } = {}): string {
  const existing = getCookie(c, ANON_COOKIE_NAME);
  if (!opts.forceReissue && existing && ULID_REGEX.test(existing)) return existing;

  const id = ulid();
  setCookie(c, ANON_COOKIE_NAME, id, {
    httpOnly: true,
    sameSite: 'Lax',
    secure: c.env.ENVIRONMENT === 'production',
    path: '/',
    maxAge: ANON_COOKIE_MAX_AGE_SEC,
  });
  return id;
}

export function isColdStartLanding(c: Context): boolean {
  // Session-fixation mitigation: re-issue cookie on cold-start landing visit
  // (no Referer OR Referer is off-origin). Only applies to GET / requests.
  const referer = c.req.header('Referer');
  if (!referer) return true;
  try {
    const refUrl = new URL(referer);
    const ourHost = new URL(c.req.url).hostname;
    return refUrl.hostname !== ourHost;
  } catch {
    return true;
  }
}

export function clearAnonSession(c: Context): void {
  setCookie(c, ANON_COOKIE_NAME, '', {
    httpOnly: true, sameSite: 'Lax', secure: c.env.ENVIRONMENT === 'production',
    path: '/', maxAge: 0,
  });
}
```

|-----------|----------|-----------|-------------|-----------------|
| T-9.1-claim | Spoofing | Session-fixation: attacker pre-sets cookie on victim | mitigate | `isColdStartLanding(c)` re-issues cookie when no Referer or off-origin Referer; vitest regression test |
| T-9.1-03-01 | Information Disclosure | User A guesses User B's `job_id` and reads stream | mitigate | Cookie-scoped read in anon-stream handler; 403 if mismatch |
| T-9.1-03-02 | Information Disclosure | XSS reads anon cookie via JS | mitigate | HttpOnly flag set on every cookie issuance; tested |
| T-9.1-03-03 | Tampering | Cookie value tampered to ULID of another session | mitigate | ULID regex check; even if attacker forges valid format, they need the EXACT 26-char value (~10^39 search space) |
| T-9.1-03-04 | Repudiation | Anon user denies generating spec | accept | Anon by design; rate-limit log (plan 09.1-05) provides forensics within 30d |
