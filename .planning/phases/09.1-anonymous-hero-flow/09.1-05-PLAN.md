```typescript
// apps/api/src/lib/anon-ip-hash.ts
import { sql } from 'drizzle-orm';
import { db } from '../db.js';

interface SaltPair { current: string; previous: string | null; }

let cachedSalts: { pair: SaltPair; expires_at: number } | null = null;
const SALT_CACHE_TTL_MS = 60_000;

export async function getCurrentSalts(): Promise<SaltPair> {
  if (cachedSalts && cachedSalts.expires_at > Date.now()) return cachedSalts.pair;
  const r = await db.execute(sql`
    SELECT key, value FROM app_secrets WHERE key IN ('daily_salt_current','daily_salt_previous')
  `);
  const rows = r.rows as { key: string; value: string }[];
  const pair: SaltPair = {
    current: rows.find((x) => x.key === 'daily_salt_current')?.value ?? '',
    previous: rows.find((x) => x.key === 'daily_salt_previous')?.value ?? null,
  };
  cachedSalts = { pair, expires_at: Date.now() + SALT_CACHE_TTL_MS };
  return pair;
}

export async function hashIpForRateLimit(rawIp: string): Promise<string> {
  const { current } = await getCurrentSalts();
  const buf = new TextEncoder().encode(`${current}:${rawIp}`);
  const digest = await crypto.subtle.digest('SHA-256', buf);
  return Array.from(new Uint8Array(digest), (b) => b.toString(16).padStart(2, '0')).join('');
}

export async function hashIpsForLookup(rawIp: string): Promise<string[]> {
  const { current, previous } = await getCurrentSalts();
  const hashes: string[] = [];
  for (const salt of [current, previous]) {
    if (!salt) continue;
    const buf = new TextEncoder().encode(`${salt}:${rawIp}`);
    const digest = await crypto.subtle.digest('SHA-256', buf);
    hashes.push(Array.from(new Uint8Array(digest), (b) => b.toString(16).padStart(2, '0')).join(''));
  }
  return hashes;
}

export function extractRawIp(c: Context): string {
  // CF-Connecting-IP is set by Cloudflare and cannot be spoofed by client
  const cfIp = c.req.header('CF-Connecting-IP');
  if (cfIp) return cfIp;
  // Fallback for local dev: x-forwarded-for first hop
  const xff = c.req.header('x-forwarded-for');
  if (xff) return xff.split(',')[0]!.trim();
  // Last fallback: localhost
  return '127.0.0.1';
}

export function isLocalhostIp(ip: string): boolean {
  return ip === '127.0.0.1' || ip === '::1' || ip.startsWith('192.168.') || ip.startsWith('10.');
}

// Test-only helper (gated via NODE_ENV === 'test'):
export function __resetSaltCache(): void { cachedSalts = null; }
```

```typescript
// apps/api/src/middleware/anon-rate-limit.ts
import { createMiddleware } from 'hono/factory';
import { sql } from 'drizzle-orm';
import { db } from '../db.js';
import { extractRawIp, isLocalhostIp, hashIpForRateLimit, hashIpsForLookup } from '../lib/anon-ip-hash.js';

export const anonRateLimit = createMiddleware(async (c, next) => {
  // Skip for authenticated requests (only anon path uses this middleware)
  if (c.var.auth?.organizationId) return next();

  const rawIp = extractRawIp(c);
  if (isLocalhostIp(rawIp)) {
    // D-02 bypass for founder/CI
    c.set('rateLimitedIpHash', await hashIpForRateLimit(rawIp));
    return next();
  }

  const ipHashes = await hashIpsForLookup(rawIp);
  const r = await db.execute(sql`
    SELECT COUNT(*)::int AS count FROM anon_generation_log
    WHERE ip_hash = ANY(${ipHashes})
      AND created_at >= NOW() - INTERVAL '24 hours'
  `);
  const count = (r.rows[0] as { count: number }).count;

  if (count >= 1) {
    return c.json({
      error: 'anon_rate_limit',
      message: 'Anonymous tier limited to 1 generation per 24 hours. Sign up for 5/month free.',
      signup_url: '/sign-up',
    }, 429);
  }

  // Stash current-salt hash on the context so the route handler can INSERT into
  // anon_generation_log + anonymous_generations.ip_hash with the SAME hash.
  c.set('rateLimitedIpHash', await hashIpForRateLimit(rawIp));
  return next();
});
```
