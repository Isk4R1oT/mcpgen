Claim handler shape (RESEARCH §9 lines 1043-1101 verbatim):

```text
import { Hono } from 'hono';
import { getCookie } from 'hono/cookie';
import { sql } from 'drizzle-orm';
import { db } from '../../db.js';
import { retagScript } from '../../lib/cf-platforms-deploy.js';
import { ANON_COOKIE_NAME, clearAnonSession } from '../../lib/anon-session.js';

export const claimRoute = new Hono();

claimRoute.post('/', async (c) => {
  const { organizationId } = c.var.auth;
  if (!organizationId) return c.json({ error: 'no_org' }, 403);

  const anonSessionId = getCookie(c, ANON_COOKIE_NAME);
  if (!anonSessionId) return c.json({ error: 'no_anon_session' }, 400);

  const result = await db.transaction(async (tx) => {
    const updated = await tx.execute(sql`
      UPDATE anonymous_generations
      SET claimed_at = NOW(), claimed_by_org_id = ${organizationId}
      WHERE anon_session_id = ${anonSessionId}
        AND claimed_at IS NULL
      RETURNING generation_id
    `);
    const claimedGenIds = (updated.rows as { generation_id: string }[]).map((r) => r.generation_id);

    const deployments = await tx.execute(sql`
      SELECT id, cf_worker_name FROM deployments
      WHERE anon_session_id = ${anonSessionId}
        AND expires_at IS NOT NULL
    `);
    const deploymentRows = deployments.rows as { id: string; cf_worker_name: string }[];

    // Re-tag THEN update DB (matches cleanup-cron race protection — see T-9.1-08-06)
    for (const d of deploymentRows) {
      await retagScript(c.env, d.cf_worker_name, [
        'anon=false',
        `claimed_by_org=${organizationId}`,
      ]);
    }
    if (deploymentRows.length > 0) {
      await tx.execute(sql`
        UPDATE deployments
        SET expires_at = NULL, anon_session_id = NULL
        WHERE anon_session_id = ${anonSessionId}
      `);
    }

    return { claimed_count: claimedGenIds.length, deployments_retagged: deploymentRows.length };
  });

  clearAnonSession(c);
  return c.json(result);
});
```

```text
import { csrf } from 'hono/csrf';

protectedApp.use('*', csrf({
  origin: (origin) => {
    if (origin === 'https://mcpgen.app') return true;
    if (/^https:\/\/mcpgen-.*\.vercel\.app$/.test(origin)) return true;
    if (process.env.NODE_ENV !== 'production' && origin === 'http://localhost:3000') return true;

  },
}));

```

| T-9.1-09-02 | Spoofing (session-fixation) | Attacker pre-sets cookie before signup, victim's signup claims attacker's gens | mitigate | Plan 09.1-03 fixation re-issue logic + this plan's transactional UPDATE WHERE claimed_at IS NULL — first claim wins; subsequent (attacker) claim returns 0 rows |
