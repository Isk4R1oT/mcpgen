```text
// apps/api/src/lib/cf-platforms-deploy.ts
interface Env {
  CF_API_TOKEN: string;
  CF_ACCOUNT_ID: string;
  ENVIRONMENT?: string;
  MCPGEN_LOCAL_COMPUTE?: string;
}

interface DeployScriptParams {
  scriptName: string;
  scriptContent: string;          // bundle from Stage E
  bindings: Record<string, unknown>;
  tags: string[];                 // ["anon=true", "expires_at=<ISO>", "session=<ULID>"]
}

const NAMESPACE = 'mcpgen-prod'; // single namespace; Pitfall #11 invariant

export async function deployScript(env: Env, params: DeployScriptParams): Promise<{ url: string }> {
  if (env.MCPGEN_LOCAL_COMPUTE === '1') {
    // Local-compute fallback: return localhost URL stub.
    return { url: `http://localhost:9000/${params.scriptName}/mcp` };
  }
  const url = `https://api.cloudflare.com/client/v4/accounts/${env.CF_ACCOUNT_ID}/workers/dispatch/namespaces/${NAMESPACE}/scripts/${params.scriptName}`;
  const metadata = {
    main_module: 'worker.mjs',
    bindings: Object.entries(params.bindings).map(([name, value]) => ({ name, type: 'plain_text', text: String(value) })),
    tags: params.tags,
    compatibility_date: '2026-04-24',
    compatibility_flags: ['nodejs_compat'],
  };
  const formData = new FormData();
  formData.append('metadata', new Blob([JSON.stringify(metadata)], { type: 'application/json' }));
  formData.append('worker.mjs', new Blob([params.scriptContent], { type: 'application/javascript+module' }));

  const res = await fetch(url, {
    method: 'PUT',
    headers: { 'Authorization': `Bearer ${env.CF_API_TOKEN}` },
    body: formData,
  });
  if (!res.ok) throw new Error(`CF deploy failed ${res.status}: ${await res.text()}`);

  return { url: `https://${params.scriptName}.mcpgen.app/mcp` };
}

export async function deleteScript(env: Env, scriptName: string): Promise<void> {
  if (env.MCPGEN_LOCAL_COMPUTE === '1') return;
  const url = `https://api.cloudflare.com/client/v4/accounts/${env.CF_ACCOUNT_ID}/workers/dispatch/namespaces/${NAMESPACE}/scripts/${scriptName}`;
  const res = await fetch(url, {
    method: 'DELETE',
    headers: { 'Authorization': `Bearer ${env.CF_API_TOKEN}` },
  });
  if (!res.ok && res.status !== 404) throw new Error(`CF delete failed ${res.status}: ${await res.text()}`);
}

export async function retagScript(env: Env, scriptName: string, newTags: string[]): Promise<void> {
  if (env.MCPGEN_LOCAL_COMPUTE === '1') return;
  const url = `https://api.cloudflare.com/client/v4/accounts/${env.CF_ACCOUNT_ID}/workers/dispatch/namespaces/${NAMESPACE}/scripts/${scriptName}/tags`;
  const res = await fetch(url, {
    method: 'PUT',
    headers: {
      'Authorization': `Bearer ${env.CF_API_TOKEN}`,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify(newTags),
  });
  if (!res.ok) throw new Error(`CF retag failed ${res.status}: ${await res.text()}`);
}
```

- script_name: `anon-{nanoid}` (lowercase, hyphen, alphanumeric)
- subdomain: `https://anon-{nanoid}.mcpgen.app/mcp`
- tag format: `["anon=true", "expires_at=2026-05-02T00:00:00.000Z", "session=01HX..."]`
- Use `=` separator (free-form CF tag strings)

| Boundary | Description |
|----------|-------------|
| BFF → CF API | CF_API_TOKEN bearer auth; rate-limited at CF; never persisted in worker |
| BFF → deployments table | Anon row tagged with anon_session_id; cookie required for status read |
| CF API → tagged scripts | Tags free-form; cleanup cron filters by `anon=true` AND `expires_at < NOW()` |
