// apps/api/src/lib/cache-replay-timeline.ts
import type { GenerationStage } from '@mcpgen/contracts';
import { ulid } from 'ulid';

export interface CacheReplayEvent {
  readonly job_id: string;
  readonly event_id: string;
  readonly stage: GenerationStage;
  readonly status: 'started' | 'completed';
  readonly partial_result?: Record<string, unknown>;
  readonly delayMs: number;
}

const STAGES: ReadonlyArray<GenerationStage> = [
  'A', 'B', 'C', 'D', 'E', 'F1', 'F2', 'F3', 'completed',
];

const CACHE_REPLAY_DELAY_MIN_MS = 40;
const CACHE_REPLAY_DELAY_MAX_MS = 120;

export interface CachedGen {
  generation_id: string;
  quality_score: number;
  created_at: Date;
  final_tools: unknown;
  quality_report: unknown;
}

export function buildCacheReplayTimeline(
  jobId: string,
  cached: CachedGen,
): ReadonlyArray<CacheReplayEvent> {
  const range = CACHE_REPLAY_DELAY_MAX_MS - CACHE_REPLAY_DELAY_MIN_MS;
  return STAGES.map((stage, idx) => {
    const isFirst = idx === 0;
    const isTerminal = stage === 'completed';

    const partial: Record<string, unknown> = isFirst
      ? {
          cache_hit: true,
          original_quality: cached.quality_score,
          original_generation_id: cached.generation_id,
          served_from: cached.created_at.toISOString(),
        }
      : {};

    if (isTerminal) {
      partial.final_tools = cached.final_tools;
      partial.quality_report = cached.quality_report;
    }

    return {
      job_id: jobId,
      event_id: ulid(),
      stage,
      status: isTerminal ? 'completed' : 'started',
      partial_result: Object.keys(partial).length > 0 ? partial : undefined,
      delayMs: CACHE_REPLAY_DELAY_MIN_MS + Math.random() * range,
    };
  });
}

// apps/api/src/lib/spec-cache-lookup.ts
import { sql } from 'drizzle-orm';
import { db } from '../db.js';

export interface CachedGenLookupResult {
  generation_id: string;
  spec_id: string;
  quality_score: number;
  quality_report: unknown;
  final_tools: unknown;
  created_at: Date;
}

export async function lookupCachedGeneration(specHash: string): Promise<CachedGenLookupResult | null> {
  const r = await db.execute(sql`
    SELECT g.id AS generation_id, g.spec_id, g.quality_score, g.quality_report,
           g.final_tools, g.created_at
    FROM generations g
    JOIN specs s ON s.id = g.spec_id
    WHERE s.content_hash = ${specHash}
      AND g.quality_score >= 4.0                       -- verified+ floor (D-05)
      AND g.is_publishable = true
      AND g.created_at >= NOW() - INTERVAL '7 days'   -- freshness (D-05)
      AND g.cached_from_generation_id IS NULL          -- Pitfall #6: never chain
    ORDER BY g.quality_score DESC, g.created_at DESC
    LIMIT 1
  `);
  return (r.rows[0] as CachedGenLookupResult | undefined) ?? null;
}

export async function createCacheHitChild(
  cached: CachedGenLookupResult,
  anonSessionId: string,
  ipHash: string,
): Promise<{ new_generation_id: string }> {
  // Atomic-ish: in a transaction, INSERT new generations row + anonymous_generations row.
  // Drizzle transaction.
  const result = await db.transaction(async (tx) => {
    const ins = await tx.execute(sql`
      INSERT INTO generations (
        spec_id, status, quality_score, quality_report, final_tools,
        cached_from_generation_id, llm_cost_usd, is_publishable, created_at
      ) VALUES (
        ${cached.spec_id}, 'completed', ${cached.quality_score},
        ${sql.raw(JSON.stringify(cached.quality_report))}::jsonb,
        ${sql.raw(JSON.stringify(cached.final_tools))}::jsonb,
        ${cached.generation_id}, 0, false, NOW()
      )
      RETURNING id
    `);
    const newGenId = (ins.rows[0] as { id: string }).id;

    await tx.execute(sql`
      INSERT INTO anonymous_generations (anon_session_id, generation_id, ip_hash)
      VALUES (${anonSessionId}, ${newGenId}, ${ipHash})
    `);
    return newGenId;
  });
  return { new_generation_id: result };
}
