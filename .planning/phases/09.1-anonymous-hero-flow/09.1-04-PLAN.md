```sql
-- 20260501010000_phase09_1_anon_flow.sql

-- 1. anonymous_generations
CREATE TABLE anonymous_generations (
  anon_session_id   TEXT PRIMARY KEY,             -- ULID
  generation_id     UUID NOT NULL REFERENCES generations(id) ON DELETE CASCADE,
  ip_hash           TEXT NOT NULL,                 -- sha256(daily_salt + raw_ip)
  created_at        TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  claimed_at        TIMESTAMPTZ,
  claimed_by_org_id UUID REFERENCES organizations(id) ON DELETE SET NULL
);
CREATE INDEX anon_gen_session_idx ON anonymous_generations(anon_session_id);
CREATE INDEX anon_gen_unclaimed_idx ON anonymous_generations(created_at) WHERE claimed_at IS NULL;

-- 2. anon_generation_log (TimescaleDB hypertable, 1-day chunks per Pitfall #4)
CREATE TABLE anon_generation_log (
  id           BIGSERIAL,
  ip_hash      TEXT NOT NULL,
  created_at   TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
SELECT create_hypertable(
  'anon_generation_log',
  'created_at',
  chunk_time_interval => INTERVAL '1 day'
);
CREATE INDEX anon_log_ip_time_idx ON anon_generation_log(ip_hash, created_at DESC);

-- 3. app_secrets — daily-rotating salt store
CREATE TABLE app_secrets (
  key          TEXT PRIMARY KEY,
  value        TEXT NOT NULL,
  created_at   TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  rotates_at   TIMESTAMPTZ
);

-- Initial seed: daily_salt_current via pgcrypto (already enabled per FND-08)
INSERT INTO app_secrets (key, value)
VALUES (
  'daily_salt_current',
  encode(gen_random_bytes(32), 'hex')
);
-- daily_salt_previous starts NULL; first rotation cron promotes current → previous.

-- 4. deployments extension (anon ephemeral support)
ALTER TABLE deployments
  ADD COLUMN anon_session_id TEXT REFERENCES anonymous_generations(anon_session_id) ON DELETE SET NULL,
  ADD COLUMN expires_at      TIMESTAMPTZ;
CREATE INDEX deploy_expires_idx ON deployments(expires_at) WHERE expires_at IS NOT NULL;

-- 5. generations.cached_from_generation_id (OQ-2 resolution per RESEARCH §6 line 800)
ALTER TABLE generations
  ADD COLUMN cached_from_generation_id UUID REFERENCES generations(id) ON DELETE SET NULL;
CREATE INDEX gen_cached_from_idx ON generations(cached_from_generation_id) WHERE cached_from_generation_id IS NOT NULL;
```

```typescript
export const anonymousGenerations = pgTable('anonymous_generations', {
  anonSessionId: text('anon_session_id').primaryKey(),
  generationId: uuid('generation_id').notNull().references(() => generations.id, { onDelete: 'cascade' }),
  ipHash: text('ip_hash').notNull(),
  createdAt: timestamp('created_at', { withTimezone: true }).notNull().defaultNow(),
  claimedAt: timestamp('claimed_at', { withTimezone: true }),
  claimedByOrgId: uuid('claimed_by_org_id').references(() => organizations.id, { onDelete: 'set null' }),
}, (t) => ({
  anonGenSessionIdx: index('anon_gen_session_idx').on(t.anonSessionId),
  anonGenUnclaimedIdx: index('anon_gen_unclaimed_idx').on(t.createdAt).where(sql`${t.claimedAt} IS NULL`),
}));

export const anonGenerationLog = pgTable('anon_generation_log', {
  id: bigserial('id', { mode: 'bigint' }),
  ipHash: text('ip_hash').notNull(),
  createdAt: timestamp('created_at', { withTimezone: true }).notNull().defaultNow(),
}, (t) => ({
  anonLogIpTimeIdx: index('anon_log_ip_time_idx').on(t.ipHash, t.createdAt.desc()),
}));

export const appSecrets = pgTable('app_secrets', {
  key: text('key').primaryKey(),
  value: text('value').notNull(),
  createdAt: timestamp('created_at', { withTimezone: true }).notNull().defaultNow(),
  rotatesAt: timestamp('rotates_at', { withTimezone: true }),
});
```
