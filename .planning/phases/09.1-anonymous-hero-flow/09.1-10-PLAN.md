```text
export const anonTenantCleanup = inngest.createFunction(
  { id: INNGEST_FUNCTION_IDS.ANON_TENANT_CLEANUP, triggers: [{ cron: '*/15 * * * *' }] },
  async ({ step }) => {
    const expired = await step.run('list-expired', async () => {
      const r = await db.execute(sql`
        SELECT id, cf_worker_name, dispatch_namespace
        FROM deployments
        WHERE expires_at IS NOT NULL
          AND expires_at < NOW()
          AND anon_session_id IS NOT NULL
        LIMIT 500
      `);
      return r.rows;
    });
    if (expired.length === 0) {
      await sendBetterStackHeartbeat();
      return { deleted: 0 };
    }
    let deleted = 0;
    for (const d of expired) {
      await step.run(`delete-cf-${d.id}`, async () => {
        await deleteScript(env, d.cf_worker_name); // 404 = success per cf-platforms-deploy.ts
      });
      await step.run(`delete-db-${d.id}`, async () => {
        await db.execute(sql`DELETE FROM deployments WHERE id = ${d.id}`);
      });
      deleted += 1;
    }
    await sendBetterStackHeartbeat();
    return { deleted };
  },
);
```

anon-rate-limit-cleanup shape (RESEARCH §4 lines 449-475 verbatim):

```text
export const anonRateLimitCleanup = inngest.createFunction(
  { id: INNGEST_FUNCTION_IDS.ANON_RATE_LIMIT_CLEANUP, triggers: [{ cron: '0 4 * * *' }] },
  async ({ step }) => {
    const dropped = await step.run('drop-old-chunks', async () => {
      const r = await db.execute(sql`SELECT drop_chunks('anon_generation_log', INTERVAL '30 days')`);
      return r.rows.length;
    });
    const deleted = await step.run('delete-old-unclaimed', async () => {
      const r = await db.execute(sql`
        DELETE FROM anonymous_generations
        WHERE claimed_at IS NULL
          AND created_at < NOW() - INTERVAL '7 days'
        RETURNING anon_session_id
      `);
      return r.rows.length;
    });
    return { dropped_chunks: dropped, deleted_sessions: deleted };
  },
);
```

anon-salt-rotate shape (RESEARCH §5 lines 522-557 verbatim — 24h grace pattern, NOT 30s per G-4):

```text
export const anonSaltRotate = inngest.createFunction(
  { id: INNGEST_FUNCTION_IDS.ANON_SALT_ROTATE, triggers: [{ cron: '0 0 * * *' }] },
  async ({ step }) => {
    const newSalt = await step.run('generate-salt', async () => {
      const bytes = new Uint8Array(32);
      crypto.getRandomValues(bytes);
      return Array.from(bytes, (b) => b.toString(16).padStart(2, '0')).join('');
    });
    await step.run('rotate-salts', async () => {
      // 1. promote current → previous
      await db.execute(sql`
        INSERT INTO app_secrets (key, value, rotates_at)
        VALUES ('daily_salt_previous', (SELECT value FROM app_secrets WHERE key = 'daily_salt_current'), NOW() + INTERVAL '24 hours')
        ON CONFLICT (key) DO UPDATE SET value = EXCLUDED.value, rotates_at = EXCLUDED.rotates_at
      `);
      // 2. write new current
      await db.execute(sql`
        INSERT INTO app_secrets (key, value, rotates_at)
        VALUES ('daily_salt_current', ${newSalt}, NOW() + INTERVAL '24 hours')
        ON CONFLICT (key) DO UPDATE SET value = EXCLUDED.value, rotates_at = EXCLUDED.rotates_at
      `);
    });
    return { rotated: true, salt_prefix: newSalt.slice(0, 8) };
  },
);
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Inngest cron → CF API | CF_API_TOKEN bearer; cleanup uses DELETE which is idempotent |
| Inngest cron → DB | Direct drizzle queries; no user input |
| Salt rotate → app_secrets | UPSERT pattern; race-safe via ON CONFLICT |
| BetterStack heartbeat → 3rd-party | Public URL; not secret but should be CI-only |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-9.1-10-01 | Availability (cost runaway) | Cleanup cron silently fails → CF script count climbs → $80/day surprise | mitigate | BetterStack heartbeat fires every 15 min; missing heartbeat → alert. RESEARCH §3 hard requirement. Runbook ensures W7 provisioning. |
| T-9.1-10-02 | Tampering | Salt rotation race — request inflight at exactly 00:00 UTC (Pitfall #5) | mitigate | 24h grace via dual-salt lookup (RESEARCH §5; G-4 default) |
| T-9.1-10-03 | Information Disclosure | Salt value logged via Inngest observability | mitigate | Return only `salt_prefix` (8 chars) in step result; full salt never logged |
| T-9.1-10-04 | Tampering | drop_chunks runs against wrong table (e.g., usage_events) due to typo | mitigate | Hardcoded literal `'anon_generation_log'`; DB role limited per CTRL-04; vitest test 1 verifies behavior |
| T-9.1-10-05 | Repudiation | Cleanup-cron deletes claimed deployments (race with claim) | mitigate | Plan 09 claim transaction sets `expires_at = NULL AND anon_session_id = NULL` simultaneously; cleanup query filters require both fields non-null. T-9.1-08-06 regression test in plan 09. |
| T-9.1-10-06 | DoS | Inngest step timeout on >500 expired tenants | mitigate | LIMIT 500 per run; next 15-min run picks up backlog; 4 backlog runs/hour clears 2K backlog/day |
