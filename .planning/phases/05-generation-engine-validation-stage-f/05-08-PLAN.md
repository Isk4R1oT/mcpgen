    ```python
    """Cascade L2 invalidation per CONTEXT D-26.

    Strict downstream-only invalidation. Conservative — over-invalidates slightly
    rather than under-invalidate and serve stale outputs (D-26).
    """
    from __future__ import annotations

    from typing import Final

    from mcpgen_engine.cache.l2 import L2Cache


    PASS_DOWNSTREAM: Final[dict[str, list[str]]] = {
        "pass_0": ["pass_1", "pass_2", "pass_3", "pass_4", "pass_5", "stage_e", "stage_f1", "stage_f2"],
        "pass_1": ["pass_2", "pass_3", "pass_4", "pass_5", "stage_e", "stage_f1", "stage_f2"],
        "pass_2": ["pass_3", "pass_4", "pass_5", "stage_e", "stage_f1", "stage_f2"],
        "pass_3": ["pass_4", "pass_5", "stage_e", "stage_f1", "stage_f2"],
        "pass_4": ["pass_5", "stage_e", "stage_f1", "stage_f2"],
        "pass_5": ["stage_e", "stage_f1", "stage_f2"],
        "stage_e": ["stage_f1", "stage_f2"],
    }


    async def invalidate_cascade(retry_target: str, l2: L2Cache) -> int:
        """Invalidate L2 entries downstream of retry_target. Returns count removed."""
        if retry_target not in PASS_DOWNSTREAM:
            return 0
        total = 0
        for downstream in PASS_DOWNSTREAM[retry_target]:
            total += await l2.invalidate_by_prefix(f"l2:{downstream}:")
        return total
    ```
    NOTE: The actual prefix string format depends on `keys.py::l2_key`'s pre-hash composition. Check `keys.py`: the raw key is `l2:{_engine_version()}:{pass_name}:{pass_version}:...`. So the prefix to match is `l2:{engine_version}:{downstream_pass}:`. Adjust the hardcoded prefix above to include engine_version dynamically (read once at module load).

    ```python
    """Retry FSM (CONTEXT D-24, D-25, D-26, D-27)."""
    from __future__ import annotations

    import logging
    import time
    from dataclasses import dataclass, field
    from enum import Enum
    from typing import Any

    from mcpgen_engine.cache.l2 import L2Cache
    from mcpgen_engine.cache.cache_invalidation import invalidate_cascade
    from mcpgen_engine.observability import LAUNCH_CRITERIA
    from .failure_patterns import (
        f1_check_to_retry, f2_component_to_retry, f3_pattern_to_retry,
    )

    log = logging.getLogger(__name__)

    _MAX_ROUNDS = 2
    _COST_CAP_FREE_USD = 0.50
    _COST_CAP_PRO_USD = 2.00
    _WALL_CLOCK_FREE_SECONDS = 600.0
    _WALL_CLOCK_PRO_SECONDS = 1800.0


    class RetryState(str, Enum):
        INITIAL = "initial"
        F1_RUNNING = "f1_running"
        F1_DONE = "f1_done"
        F2_RUNNING = "f2_running"
        F2_DONE = "f2_done"
        F3_RUNNING = "f3_running"
        F3_DONE = "f3_done"
        RETRY_PLANNED = "retry_planned"
        UPSTREAM_RUNNING = "upstream_running"
        VALIDATION_COMPLETE = "validation_complete"
        TERMINAL_FAILURE = "terminal_failure"


    @dataclass
    class RetryTrigger:
        source: str          # f1_check_name | f2_component | f3_pattern
        target_pass: str     # pass_0/1/2/3/4/5/stage_e
        reason: str


    @dataclass
    class RetryContext:
        state: RetryState = RetryState.INITIAL
        round: int = 1
        cumulative_cost_usd: float = 0.0
        wall_clock_start: float = field(default_factory=time.time)
        triggers: list[RetryTrigger] = field(default_factory=list)
        history: list[dict[str, Any]] = field(default_factory=list)
        is_pro_tier: bool = False

        def cumulative_wall_clock(self) -> float:
            return time.time() - self.wall_clock_start


    def can_retry(ctx: RetryContext) -> tuple[bool, str | None]:
        """Returns (allowed, reason_if_blocked)."""
        if ctx.round > _MAX_ROUNDS:
            return False, "RETRY_ROUNDS_EXHAUSTED"
        cap = _COST_CAP_PRO_USD if ctx.is_pro_tier else _COST_CAP_FREE_USD
        if ctx.cumulative_cost_usd >= cap:
            return False, "RETRY_BUDGET_EXHAUSTED_COST"
        wall = _WALL_CLOCK_PRO_SECONDS if ctx.is_pro_tier else _WALL_CLOCK_FREE_SECONDS
        if ctx.cumulative_wall_clock() >= wall:
            return False, "RETRY_BUDGET_EXHAUSTED_TIME"
        return True, None


    def plan_f1_retry(error_code: str, ctx: RetryContext) -> RetryTrigger | None:
        target = f1_check_to_retry(error_code)
        if target is None:
            return None  # terminal — no retry path
        return RetryTrigger(source=f"f1.{error_code}", target_pass=target, reason=error_code)


    def plan_f2_retry(per_component_failures: dict[str, float], ctx: RetryContext) -> list[RetryTrigger]:
        triggers: list[RetryTrigger] = []
        for component, score in per_component_failures.items():
            if score >= 3.0:
                continue
            target = f2_component_to_retry(component)
            if target is None:
                continue
            triggers.append(RetryTrigger(
                source=f"f2.{component}", target_pass=target,
                reason=f"score {score:.2f} < 3.0",
            ))
        # Deduplicate by target_pass (one retry call per pass)
        by_pass: dict[str, RetryTrigger] = {}
        for t in triggers:
            by_pass.setdefault(t.target_pass, t)
        return list(by_pass.values())


    def plan_f3_retry(failure_patterns: list[str], ctx: RetryContext) -> list[RetryTrigger]:
        triggers: list[RetryTrigger] = []
        for pat in failure_patterns:
            target = f3_pattern_to_retry(pat)
            if target is None:
                continue
            triggers.append(RetryTrigger(source=f"f3.{pat}", target_pass=target, reason=pat))
        by_pass: dict[str, RetryTrigger] = {}
        for t in triggers:
            by_pass.setdefault(t.target_pass, t)
        return list(by_pass.values())


    async def execute_retry_cascade(triggers: list[RetryTrigger], l2: L2Cache) -> int:
        """Cascade L2 invalidation for each unique target_pass per CONTEXT D-26."""
        total = 0
        # Find earliest pass to invalidate (everything downstream gets cleared anyway)
        target_set = {t.target_pass for t in triggers}
        # Order matters: invalidate the EARLIEST pass first → cascades naturally
        order = ["pass_0", "pass_1", "pass_2", "pass_3", "pass_4", "pass_5", "stage_e"]
        for pass_name in order:
            if pass_name in target_set:
                total += await invalidate_cascade(pass_name, l2)
                break  # single earliest pass cascades downstream
        return total
    ```

    ```python
    """QualityReport composite score + badge assembly (CONTEXT D-28, D-29).

    Formula:
      if F1 fails: overall=0.0; badge=needs_review
      elif F2 fails AND no F3: overall=2.5; badge=needs_review
      elif F2 passes AND no F3: overall=0.5*F2/5 + 0.5; badge=standard or verified
      else (all 3 ran): overall = 0.10*F1.passed + 0.40*(F2/5) + 0.50*F3.pass_rate
          badge = premium / verified / standard / needs_review per thresholds
    """
    from __future__ import annotations

    from dataclasses import dataclass
    from typing import Any

    from mcpgen_engine.observability import LAUNCH_CRITERIA


    def compute_overall(*, f1_passed: bool, f2_overall: float | None,
                        f3_pass_rate: float | None) -> float:
        """Returns 0–5 score per D-28 (formula × 5)."""
        if not f1_passed:
            return 0.0
        if f3_pass_rate is None:
            # F2 only
            if f2_overall is None or f2_overall < LAUNCH_CRITERIA["F2_SMELL_MIN"]:
                return 2.5
            return (0.5 * f2_overall / 5 + 0.5) * 5
        # All 3 ran
        f2_norm = (f2_overall or 0.0) / 5.0
        score = 0.10 * 1.0 + 0.40 * f2_norm + 0.50 * f3_pass_rate
        return score * 5


    def compute_badge(*, f1_passed: bool, f2_overall: float | None,
                      f3_pass_rate: float | None) -> str:
        """Badge per D-28 thresholds."""
        if not f1_passed:
            return "needs_review"
        f2 = f2_overall or 0.0
        f2_min = LAUNCH_CRITERIA["F2_SMELL_MIN"]
        f3_min = LAUNCH_CRITERIA["F3_AGENT_PASS_RATE_MIN"]
        if f3_pass_rate is None:
            if f2 >= f2_min:
                return "verified"  # technically D-28 says verified requires F3>=0.7 too;
                # but path "F2 passes AND no F3" maps to standard/verified — clarify D-28 wording
                # treats verified as F1+F2>=4.0 + (F3 absent OR >=0.7). Per D-28: standard if F2>=3.5; verified if F2>=4.0 + F3>=0.7. With no F3, standard is the safest verdict per D-28's "standard if F1.passed and F2>=3.5 and (no F3 OR F3>=0.5)".
            if f2 >= 3.5:
                return "standard"
            return "needs_review"
        # All 3 ran
        if f1_passed and f2 >= 4.5 and f3_pass_rate >= 0.85:
            return "premium"
        if f1_passed and f2 >= f2_min and f3_pass_rate >= f3_min:
            return "verified"
        if f1_passed and f2 >= 3.5 and f3_pass_rate >= 0.5:
            return "standard"
        return "needs_review"
    ```
    NOTE on the F2-without-F3 case: D-28 says `else if F2 passes AND no F3 (opt-out, σ ≥ 0.4): overall = 0.5 * F2.overall_score / 5 + 0.5; badge = standard or verified`. The "standard or verified" disjunction needs a deterministic rule. Using D-28 formula's "verified if F1.passed and F2 >= 4.0 and F3.pass_rate >= 0.7" — F3 absent fails the F3 condition, so the badge cannot be "verified" without F3. Use `standard` as the safest interpretation when F2 passes but F3 was opt-out. Document this inline. If clarification is needed, surface a `TODO(D-28)` comment for Plan 05-09 calibration.
