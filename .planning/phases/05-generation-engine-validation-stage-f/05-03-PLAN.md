    from __future__ import annotations

    import re
    from dataclasses import dataclass
    from pathlib import Path

    _TEMPLATE_ARTIFACT = re.compile(r"\{\{|\}\}")


    @dataclass(frozen=True)
    class TemplateArtifactResult:
        passed: bool
        error: str | None
        offending_files: list[str]


    def check_template_artifacts(generated_dir: Path) -> TemplateArtifactResult:

    Step 6: Create `f1_checks/smart_id_fuzz.py`. Pure-Python smart-ID parser (~10 lines, mirrors RESEARCH.md §3.6):

    """F1 check #3: smart-ID cross-tenant fuzz (CONTEXT D-05 step 3, Pitfall #1).

    Synthesizes 2 tenants from same spec; verifies parser rejects cross-tenant IDs.
    Pure-Python port of runtime/smart_id.ts to avoid node-bridge complexity (RESEARCH §3.6).
    """
    from __future__ import annotations

    import re
    from dataclasses import dataclass

    # Smart-ID format: {tenant_short_id}-{spec_slug}:{type}:{collection}:{identifier}
    _SMART_ID_RE = re.compile(
        r"^(?P<tenant>[a-z0-9]+-[a-z0-9_-]+):(?P<type>[a-z_]+):(?P<collection>[A-Za-z0-9_]+):(?P<id>.+)$"
    )


    @dataclass(frozen=True)
    class SmartIdParseResult:
        accepted: bool
        tenant: str | None


    def parse_smart_id(smart_id: str, *, expected_prefix: str) -> SmartIdParseResult:
        m = _SMART_ID_RE.match(smart_id)
        if m is None:
            return SmartIdParseResult(accepted=False, tenant=None)
        if m.group("tenant") != expected_prefix:
            return SmartIdParseResult(accepted=False, tenant=m.group("tenant"))
        return SmartIdParseResult(accepted=True, tenant=m.group("tenant"))


    @dataclass(frozen=True)
    class SmartIdFuzzResult:
        passed: bool
        error: str | None
        details: str


    def smart_id_fuzz(*, spec_slug: str, sample_collection: str, sample_id: str) -> SmartIdFuzzResult:
        tenant_a = f"abc1-{spec_slug}"
        tenant_b = f"xyz2-{spec_slug}"
        id_a = f"{tenant_a}:object:{sample_collection}:{sample_id}"
        id_b = f"{tenant_b}:object:{sample_collection}:{sample_id}"
        # Tenant A's parser must accept its own ID.
        own = parse_smart_id(id_a, expected_prefix=tenant_a)
        # Tenant A's parser must reject Tenant B's ID.
        cross = parse_smart_id(id_b, expected_prefix=tenant_a)
        if not own.accepted or cross.accepted:
            return SmartIdFuzzResult(passed=False, error="SMART_ID_CROSS_TENANT_LEAK",
                                     details=f"own={own.accepted}, cross={cross.accepted}")
        return SmartIdFuzzResult(passed=True, error=None, details="OK")
    ```

    """F1 check #4: MCP protocol compliance (CONTEXT D-05 step 4)."""
    from __future__ import annotations

    import json
    from dataclasses import dataclass
    from typing import Any


    @dataclass(frozen=True)
    class McpComplianceResult:
        passed: bool
        error: str | None
        offending_tools: list[tuple[str, str]]  # (tool_name, reason)


    REQUIRED_ANNOTATIONS = ("readOnlyHint", "destructiveHint", "idempotentHint", "openWorldHint")
    REQUIRED_PROTOCOL_VERSION = "2025-06-18"


    def check_mcp_compliance(*, final_tools: list[dict[str, Any]],
                             mcp_protocol_version: str) -> McpComplianceResult:
        if mcp_protocol_version != REQUIRED_PROTOCOL_VERSION:
            return McpComplianceResult(
                passed=False, error="MCP_COMPLIANCE_FAIL_PROTOCOL_VERSION",
                offending_tools=[("__manifest__", f"got {mcp_protocol_version}, want {REQUIRED_PROTOCOL_VERSION}")],
            )
        offenders: list[tuple[str, str]] = []
        for tool in final_tools:
            name = tool.get("name", "<unnamed>")
            ann = tool.get("annotations") or {}
            for key in REQUIRED_ANNOTATIONS:
                if key not in ann:
                    offenders.append((name, f"missing annotation {key}"))
            if ann.get("openWorldHint") is not True:
                offenders.append((name, "openWorldHint invariant violated (must be true)"))
            # tools/list serializability check
            try:
                json.dumps(tool, default=str)
            except (TypeError, ValueError) as e:
                offenders.append((name, f"non-serializable: {e}"))
        if offenders:
            return McpComplianceResult(passed=False, error="MCP_COMPLIANCE_FAIL_ANNOTATIONS",
                                       offending_tools=offenders[:20])
        return McpComplianceResult(passed=True, error=None, offending_tools=[])
    ```
