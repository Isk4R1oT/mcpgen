    """F3 wrangler dev --local subprocess management (CONTEXT D-18, D-51).

    Subprocess gotchas (RESEARCH §5.4):
      1. Process-group SIGTERM cascades to Miniflare workerd children.
      2. Phase 4 D-39 pre-warmed node_modules → ~3-5s startup vs ~30s cold.
      3. Port-collision retry up to 3 attempts.
      4. MCPGEN_F3_TEST=1 scoped ONLY to subprocess env (D-51 DNS bypass).
      5. Health check via tools/list JSON-RPC (no /health endpoint).
    """
    from __future__ import annotations

    import asyncio
    import contextlib
    import logging
    import os
    import signal
    import socket
    from contextlib import asynccontextmanager
    from pathlib import Path
    from typing import AsyncIterator

    import httpx

    log = logging.getLogger(__name__)

    _STARTUP_TIMEOUT = 30.0  # seconds
    _CLEANUP_TIMEOUT = 5.0
    _PORT_RETRY_MAX = 3


    def _find_free_port() -> int:
        """Bind/release pattern; small race window acceptable (RESEARCH §5.4 #3)."""
        with socket.socket() as s:
            s.bind(("127.0.0.1", 0))
            return s.getsockname()[1]


    async def _wait_until_ready(server_url: str, timeout_seconds: float) -> None:
        """Poll tools/list JSON-RPC until 200 response (RESEARCH §5.5)."""
        deadline = asyncio.get_event_loop().time() + timeout_seconds
        body = {"jsonrpc": "2.0", "id": 1, "method": "tools/list"}
        async with httpx.AsyncClient(timeout=httpx.Timeout(5.0, connect=2.0)) as client:
            while asyncio.get_event_loop().time() < deadline:
                try:
                    resp = await client.post(server_url, json=body, headers={
                        "Content-Type": "application/json",
                        "Accept": "application/json, text/event-stream",
                    })
                    if resp.status_code == 200:
                        return
                except (httpx.ConnectError, httpx.ReadError, httpx.TimeoutException):
                    pass
                await asyncio.sleep(0.5)
        raise TimeoutError(f"Server at {server_url} not ready within {timeout_seconds}s")


    async def _kill_process_group(proc: asyncio.subprocess.Process) -> None:
        """Cross-platform process-group kill cascading to subprocess children (RESEARCH §5.4 #1).

        Uses POSIX setsid + killpg pattern. On Windows, falls back to proc.terminate.
        """
        if proc.returncode is not None:
            return
        try:
            if hasattr(os, "killpg") and hasattr(os, "getpgid"):
                os.killpg(os.getpgid(proc.pid), signal.SIGTERM)
            else:
                proc.terminate()
            try:
                await asyncio.wait_for(proc.wait(), timeout=_CLEANUP_TIMEOUT)
            except asyncio.TimeoutError:
                if hasattr(os, "killpg"):
                    os.killpg(os.getpgid(proc.pid), signal.SIGKILL)
                else:
                    proc.kill()
                await proc.wait()
        except ProcessLookupError:
            pass


    @asynccontextmanager
    async def spawn_server(generated_dir: Path) -> AsyncIterator[str]:
        """Spawn `wrangler dev --local --port {N}`; yield server_url.

        D-51: MCPGEN_F3_TEST=1 set ONLY in subprocess env (not parent process env).
        """
        port = _find_free_port()
        # Subprocess-only env — parent process env unchanged.
        env = {**os.environ, "MCPGEN_F3_TEST": "1"}
        last_error: Exception | None = None

        for attempt in range(_PORT_RETRY_MAX):
            try:
                proc = await asyncio.create_subprocess_exec(
                    "npx", "wrangler", "dev",
                    "--local",
                    "--port", str(port),
                    "--ip", "127.0.0.1",
                    cwd=str(generated_dir),
                    env=env,
                    stdout=asyncio.subprocess.PIPE,
                    stderr=asyncio.subprocess.PIPE,
                    # POSIX-only: setsid → new process group for clean cascade kill.
                    start_new_session=True,
                )
                server_url = f"http://127.0.0.1:{port}"
                try:
                    await _wait_until_ready(server_url, _STARTUP_TIMEOUT)
                except TimeoutError:
                    await _kill_process_group(proc)
                    last_error = TimeoutError(f"wrangler dev startup timeout on port {port}")
                    port = _find_free_port()
                    continue
                break
            except OSError as e:
                last_error = e
                port = _find_free_port()
                continue
        else:
            raise RuntimeError(f"Failed to spawn wrangler after {_PORT_RETRY_MAX} attempts: {last_error}")

        try:
            yield server_url
        finally:
            await _kill_process_group(proc)
            # Verify MCPGEN_F3_TEST never leaked to parent
            assert "MCPGEN_F3_TEST" not in os.environ, "DNS bypass flag leaked to parent process — security violation"
    ```

    """F3 test-agent harness (CONTEXT D-19).

    Real Sonnet 4.5 (the Override doc §7.3 exception). Drives multi-turn agent
    loop driven by stop_reason; records full trajectory for QualityReport debug.
    """
    from __future__ import annotations

    import asyncio
    import json
    import logging
    from dataclasses import dataclass, field
    from typing import Any

    import httpx
    from anthropic import APIStatusError, RateLimitError
    from anthropic.types import MessageParam, ToolUseBlock
    from tenacity import (
        retry,
        retry_if_exception_type,
        stop_after_attempt,
        wait_exponential,
    )

    from mcpgen_engine.llm.test_agent import ANTHROPIC, SONNET_MODEL_ID
    from mcpgen_engine.llm.sampling import F3_TEST_AGENT_SETTINGS

    log = logging.getLogger(__name__)

    _AGENT_CONCURRENCY = 3  # Anthropic per-org rate limits (CONTEXT D-19)


    @dataclass(frozen=True)
    class ToolCall:
        tool_use_id: str
        tool_name: str
        args: dict[str, Any]


    @dataclass(frozen=True)
    class TrajectoryStep:
        turn: int
        stop_reason: str
        text_content: str
        tool_calls: list[ToolCall]


    @dataclass(frozen=True)
    class TaskTrajectory:
        steps: list[TrajectoryStep]
        tool_calls: list[ToolCall]            # flattened
        iteration_count: int
        terminated: str                        # end_turn / tool_use / max_iterations / max_tokens / pause_turn / stop_sequence
        final_text: str | None


    def mcp_tool_to_anthropic(mcp_tool: dict[str, Any]) -> dict[str, Any]:
        """Mechanical: MCP `inputSchema` → Anthropic `input_schema` (RESEARCH §5.3)."""
        return {
            "name": mcp_tool["name"],
            "description": _render_description_for_agent(mcp_tool.get("description")),
            "input_schema": mcp_tool.get("inputSchema", {}),
        }


    def _render_description_for_agent(description: dict[str, Any] | str | None) -> str:
        if description is None:
            return ""
        if isinstance(description, str):
            return description
        # Concatenate Pass 2 description components in a stable order
        parts: list[str] = []
        for key in ("purpose", "guidelines", "limitations", "parameter_overview"):
            val = description.get(key)
            if val:
                parts.append(f"## {key}\n{val}")
        return "\n\n".join(parts)


    async def _execute_mcp_tool_call(server_url: str, tool_use: ToolUseBlock,
                                      sandbox_credentials: dict[str, str] | None) -> dict[str, Any]:
        body = {
            "jsonrpc": "2.0", "id": 1, "method": "tools/call",
            "params": {"name": tool_use.name, "arguments": tool_use.input},
        }
        headers = {
            "Content-Type": "application/json",
            "Accept": "application/json, text/event-stream",
        }
        # Pass-through credentials per Pass-Through credentials default
        if sandbox_credentials and tool_use.name in sandbox_credentials:
            headers["X-Upstream-Auth"] = sandbox_credentials[tool_use.name]
        async with httpx.AsyncClient(timeout=httpx.Timeout(30.0, connect=5.0)) as client:
            resp = await client.post(server_url, json=body, headers=headers)
            return resp.json()


    @retry(
        stop=stop_after_attempt(3),
        wait=wait_exponential(multiplier=2, min=2, max=10),
        retry=retry_if_exception_type((APIStatusError, RateLimitError, httpx.NetworkError)),
        reraise=True,
    )
    async def _agent_step(messages: list[MessageParam], anthropic_tools: list[dict]) -> Any:
        return await ANTHROPIC.messages.create(
            model=SONNET_MODEL_ID,
            max_tokens=F3_TEST_AGENT_SETTINGS["max_tokens"],
            temperature=F3_TEST_AGENT_SETTINGS["temperature"],
            top_p=F3_TEST_AGENT_SETTINGS["top_p"],
            tools=anthropic_tools,
            messages=messages,
        )


    async def run_golden_task(
        *,
        task: dict[str, Any],   # GoldenTask Pydantic dict (Plan 05-01)
        server_url: str,
        mcp_tools: list[dict[str, Any]],
        sandbox_credentials: dict[str, str] | None = None,
    ) -> TaskTrajectory:
        """Multi-turn Sonnet loop. Drive via stop_reason (RESEARCH §5.1)."""
        anthropic_tools = [mcp_tool_to_anthropic(t) for t in mcp_tools]
        messages: list[MessageParam] = [{"role": "user", "content": task["prompt"]}]
        steps: list[TrajectoryStep] = []
        all_tool_calls: list[ToolCall] = []
        max_iter = task.get("max_iterations", 10)

        for turn in range(max_iter):
            resp = await _agent_step(messages, anthropic_tools)
            text_content = ""
            step_tool_calls: list[ToolCall] = []
            for block in resp.content:
                btype = getattr(block, "type", "")
                if btype == "text":
                    text_content += getattr(block, "text", "")
                elif btype == "tool_use":
                    tc = ToolCall(
                        tool_use_id=block.id,
                        tool_name=block.name,
                        args=block.input if isinstance(block.input, dict) else {},
                    )
                    step_tool_calls.append(tc)
                    all_tool_calls.append(tc)
            steps.append(TrajectoryStep(
                turn=turn, stop_reason=resp.stop_reason,
                text_content=text_content, tool_calls=step_tool_calls,
            ))

            if resp.stop_reason == "end_turn":
                return TaskTrajectory(steps=steps, tool_calls=all_tool_calls,
                                      iteration_count=turn + 1, terminated="end_turn",
                                      final_text=text_content)
            if resp.stop_reason == "tool_use":
                # Execute every tool_use against MCP server
                tool_results: list[dict[str, Any]] = []
                for block in resp.content:
                    if getattr(block, "type", "") == "tool_use":
                        result = await _execute_mcp_tool_call(server_url, block, sandbox_credentials)
                        # Unpack JSON-RPC result wrapper
                        content = result.get("result", {}).get("content") or result.get("error") or {}
                        tool_results.append({
                            "type": "tool_result",
                            "tool_use_id": block.id,
                            "content": json.dumps(content)[:5000],  # cap per call
                        })
                messages.append({"role": "assistant", "content": resp.content})
                messages.append({"role": "user", "content": tool_results})
                continue
            # Other stop_reasons: max_tokens / pause_turn / stop_sequence
            return TaskTrajectory(steps=steps, tool_calls=all_tool_calls,
                                  iteration_count=turn + 1, terminated=resp.stop_reason,
                                  final_text=text_content)

        return TaskTrajectory(steps=steps, tool_calls=all_tool_calls,
                              iteration_count=max_iter, terminated="max_iterations",
                              final_text=None)
    ```
