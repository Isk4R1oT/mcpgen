    """F1 check #9: gitleaks secret scan (CONTEXT D-05 step 9, Pitfall #12).

    SECRETS_LEAKED is TERMINAL per D-06 — operator must intervene; no retry path.
    """
    from __future__ import annotations

    import asyncio
    import json
    import shutil
    from dataclasses import dataclass
    from pathlib import Path


    @dataclass(frozen=True)
    class SecretScanResult:
        passed: bool
        error: str | None
        findings: list[dict]
        details: str


    async def run_secret_scan(generated_dir: Path, *, timeout_seconds: float = 10.0) -> SecretScanResult:
        """Invoke gitleaks 8.x with --no-git --redact (RESEARCH §3.3)."""
        if shutil.which("gitleaks") is None:
            return SecretScanResult(
                passed=False, error="GITLEAKS_NOT_INSTALLED",
                findings=[], details="gitleaks binary not on PATH; install via brew install gitleaks",
            )
        proc = await asyncio.create_subprocess_exec(
            "gitleaks", "detect",
            "--source", str(generated_dir),
            "--no-git",
            "--redact",
            "--report-format", "json",
            "--report-path", "/dev/stdout",
            "--exit-code", "0",
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE,
        )
        try:
            stdout, _stderr = await asyncio.wait_for(proc.communicate(), timeout=timeout_seconds)
        except asyncio.TimeoutError:
            proc.kill()
            await proc.wait()
            return SecretScanResult(passed=False, error="SECRET_SCAN_TIMEOUT", findings=[],
                                    details=f"gitleaks exceeded {timeout_seconds}s")

        try:
            findings = json.loads(stdout.decode("utf-8")) if stdout.strip() else []
        except json.JSONDecodeError:
            return SecretScanResult(passed=False, error="SECRET_SCAN_OUTPUT_INVALID", findings=[],
                                    details="gitleaks produced non-JSON output")

        if findings:
            return SecretScanResult(
                passed=False, error="SECRETS_LEAKED", findings=findings[:20],
                details=f"{len(findings)} potential secret(s) detected (redacted)",
            )
        return SecretScanResult(passed=True, error=None, findings=[], details="OK")
    ```

    Validates both inputSchema and outputSchema for every tool against the pinned

    from dataclasses import dataclass
    from functools import lru_cache
    from pathlib import Path
    from typing import Any

    from jsonschema import Draft202012Validator, FormatChecker


    @dataclass(frozen=True)
    class JsonSchemaError:
        tool_name: str
        kind: str  # "inputSchema" | "outputSchema"
        path: list[str | int]
        message: str


    @dataclass(frozen=True)
    class JsonSchemaResult:
        passed: bool
        error: str | None
        errors: list[JsonSchemaError]


    @lru_cache(maxsize=1)
    def _load_mcp_schema() -> dict[str, Any]:
        """Load pinned MCP 2025-06-18 schema bundle (Plan 05-02)."""
        # Walk up from this file to the repo root; locate _canonical/mcp-schema.json
        path = Path(__file__).resolve()
        for _ in range(10):
            candidate = path.parent / "packages" / "engine-fixtures" / "_canonical" / "mcp-schema.json"
            if candidate.exists():
                return json.loads(candidate.read_text(encoding="utf-8"))
            path = path.parent
        raise FileNotFoundError("packages/engine-fixtures/_canonical/mcp-schema.json not found")


    def validate_tool_schemas(final_tools: list[dict[str, Any]]) -> JsonSchemaResult:
        """Per CONTEXT D-05 step 10: validate inputSchema + outputSchema for every tool."""
        # NOTE: MCP schema bundle is for the WIRE protocol (tool definition shape).
        # We don't validate inputSchema-as-instance against MCP root; we use Draft202012
        # to assert each inputSchema/outputSchema is itself a valid Draft 2020-12 schema.
        meta_schema = Draft202012Validator.META_SCHEMA
        validator = Draft202012Validator(meta_schema, format_checker=FormatChecker())
        errors: list[JsonSchemaError] = []
        for tool in final_tools:
            name = tool.get("name", "<unnamed>")
            for kind in ("inputSchema", "outputSchema"):
                schema = tool.get(kind)
                if schema is None:
                    if kind == "outputSchema":
                        # outputSchema is OPTIONAL in MCP 2025-06-18; not present is OK
                        continue
                    errors.append(JsonSchemaError(name, kind, [], "schema is null"))
                    continue
                for err in validator.iter_errors(schema):
                    errors.append(JsonSchemaError(name, kind, list(err.absolute_path), err.message))
        if errors:
            # Differentiate input vs output for retry mapping
            has_input = any(e.kind == "inputSchema" for e in errors)
            error_code = "JSON_SCHEMA_INVALID_INPUT" if has_input else "JSON_SCHEMA_INVALID_OUTPUT"
            return JsonSchemaResult(passed=False, error=error_code, errors=errors[:20])
        return JsonSchemaResult(passed=True, error=None, errors=[])

    """F1 check #11: TypeScript compile (CONTEXT D-05 step 11)."""
    from __future__ import annotations

    import asyncio
    import re
    from dataclasses import dataclass
    from pathlib import Path

    _TSC_ERROR_RE = re.compile(
        r"^(?P<file>[^()]+)\((?P<line>\d+),(?P<col>\d+)\):\s+error\s+(?P<code>TS\d+):\s+(?P<message>.+)$"
    )


    @dataclass(frozen=True)
    class TsError:
        file: str
        line: int
        col: int
        code: str
        message: str


    @dataclass(frozen=True)
    class TsCompileResult:
        passed: bool
        error: str | None
        errors: list[TsError]


    def parse_tsc_output(stdout: str) -> list[TsError]:
        result: list[TsError] = []
        for line in stdout.splitlines():
            m = _TSC_ERROR_RE.match(line)
            if m:
                result.append(TsError(
                    file=m.group("file").strip(),
                    line=int(m.group("line")),
                    col=int(m.group("col")),
                    code=m.group("code"),
                    message=m.group("message"),
                ))
        return result


    async def run_ts_compile(generated_dir: Path, *, timeout_seconds: float = 30.0) -> TsCompileResult:
        """Invoke `npx tsc --noEmit -p tsconfig.json` (CONTEXT D-05 step 11).

        Reuses Phase 4 D-39 pre-warmed packages/codegen-templates/node_modules/.
        """
        proc = await asyncio.create_subprocess_exec(
            "npx", "tsc", "--noEmit", "-p", "tsconfig.json",
            cwd=str(generated_dir),
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE,
        )
        try:
            stdout, stderr = await asyncio.wait_for(proc.communicate(), timeout=timeout_seconds)
        except asyncio.TimeoutError:
            proc.kill()
            await proc.wait()
            return TsCompileResult(passed=False, error="TS_COMPILE_TIMEOUT", errors=[])

        # tsc writes errors to stdout (with --pretty disabled) or stderr (--pretty); capture both
        combined = (stdout.decode("utf-8") + "\n" + stderr.decode("utf-8")).strip()
        errors = parse_tsc_output(combined)[:50]  # D-05 step 11: first-50 errors only
        if proc.returncode != 0 or errors:
            return TsCompileResult(passed=False, error="TS_COMPILE_FAILED", errors=errors)
        return TsCompileResult(passed=True, error=None, errors=[])
    ```
