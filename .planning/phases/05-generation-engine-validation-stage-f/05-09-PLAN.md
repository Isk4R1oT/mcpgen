    ```typescript
    import { readFileSync, existsSync } from 'node:fs';
    import { parse as parseYaml } from 'yaml';   // pinned version in package.json

    export function loadSandboxCredentials(path: string): Record<string, string> {
      // Reject raw credentials passed directly
      if (path.startsWith('sk_') || path.startsWith('ghp_') || path.startsWith('Bearer ')) {
        throw new Error(
          'Refusing to accept raw credential as --sandbox-creds value. ' +
          'Pass a file path (e.g., --sandbox-creds ~/.mcpgen/sandbox-creds.yaml). ' +
          'See CONTEXT D-39: never via shell history.'
        );
      }
      if (!existsSync(path)) {
        throw new Error(`--sandbox-creds file not found: ${path}`);
      }
      const content = readFileSync(path, 'utf-8');
      // Support both YAML and .env-style; sniff by file extension
      if (path.endsWith('.yaml') || path.endsWith('.yml')) {
        const parsed = parseYaml(content);
        if (typeof parsed !== 'object' || parsed === null) {
          throw new Error('--sandbox-creds YAML must be a key-value object');
        }
        return parsed as Record<string, string>;
      }
      // .env-style: KEY=VALUE\n
      const result: Record<string, string> = {};
      for (const line of content.split('\n')) {
        const trimmed = line.trim();
        if (!trimmed || trimmed.startsWith('#')) continue;
        const eq = trimmed.indexOf('=');
        if (eq < 0) continue;
        result[trimmed.slice(0, eq).trim()] = trimmed.slice(eq + 1).trim();
      }
      return result;
    }
    ```
