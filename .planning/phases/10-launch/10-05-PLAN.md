```typescript
// apps/api/src/inngest/functions/usage-reconciler-v1.ts — Phase 10 wires drift webhook
// Existing reconciler reads reconciliation_log; for each tenant, compute drift = |timescale_count - stripe_count| / stripe_count
// NEW: when drift > 0.005 → POST to BETTERSTACK_INCIDENT_WEBHOOK_URL with summary + tenant_id + counts
const DRIFT_THRESHOLD = 0.005; // 0.5%
if (Math.abs(timescale - stripe) / stripe > DRIFT_THRESHOLD) {
  await fetch(env.BETTERSTACK_INCIDENT_WEBHOOK_URL, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      summary: `Usage reconciliation drift > 0.5% for tenant ${tenant_id}`,
      description: `timescale=${timescale}, stripe=${stripe}, drift=${(...).toFixed(4)}`,
      severity: 'critical',
    }),
  });
}
```
