```typescript
// Phase 10 D-03 step 7 — real-Sentry leak audit
import type { SentryEventsAdapter, SentryEvent } from './sentry-events-adapter.js';

export class RealSentryEventsAdapter implements SentryEventsAdapter {
  constructor(
    private readonly org: string,
    private readonly authToken: string,
  ) {}

  async query(pattern: string, sinceISO: string): Promise<SentryEvent[]> {
    const url = `https://sentry.io/api/0/organizations/${this.org}/events/?query=${encodeURIComponent(pattern)}&statsPeriod=24h`;
    const r = await fetch(url, {
      headers: { Authorization: `Bearer ${this.authToken}` },
    });
    if (!r.ok) throw new Error(`Sentry events API ${r.status}: ${await r.text()}`);
    const json = await r.json() as { data: SentryEvent[] };
    return json.data;
  }
}
```
